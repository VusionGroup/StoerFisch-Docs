199 struct alignas(8) TelemetryRecord {

207 template<typename T>
208 void setPayload(RecordType rt, std::uint32_t seq, std::uint64_t ts_ns, const T &value) {
209 static_assert(sizeof(T) <= kPayloadSize, "Payload zu groß für Slot");
210 timestamp_ns = ts_ns;
211 type = static_cast<std::uint32_t>(rt);
212 sequence = seq;
213 std::memset(payload, 0, kPayloadSize);
214 std::memcpy(payload, &value, sizeof(T));
215 }
216 };
217
