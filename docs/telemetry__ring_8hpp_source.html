279// Die gesamte Cross-Language-Konstruktion steht und fällt damit, dass
280// std::atomic<T> byte-identisch zu T ist und ohne versteckten Lock
281// auskommt. Auf x86-64 mit gcc/clang ist beides gegeben -- hier explizit
282// geprüft, damit ein Portierungsversuch auf eine exotische Plattform am
283// Compiler scheitert und nicht erst im Feld.
284static_assert(sizeof(std::atomic<std::uint32_t>) == sizeof(std::uint32_t),
285 "std::atomic<uint32_t> ist nicht byte-identisch zu uint32_t");
286static_assert(sizeof(std::atomic<std::uint64_t>) == sizeof(std::uint64_t),
287 "std::atomic<uint64_t> ist nicht byte-identisch zu uint64_t");
288static_assert(std::atomic<std::uint32_t>::is_always_lock_free,
289 "std::atomic<uint32_t> ist nicht lock-free -- Python könnte "
290 "den Wert dann nicht direkt aus dem Speicher lesen");
291static_assert(std::atomic<std::uint64_t>::is_always_lock_free,
292 "std::atomic<uint64_t> ist nicht lock-free -- Python könnte "
293 "den Wert dann nicht direkt aus dem Speicher lesen");
