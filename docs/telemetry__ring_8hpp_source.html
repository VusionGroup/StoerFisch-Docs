201 template <typename T>
202 void setPayload(RecordType rt, std::uint32_t seq, std::uint64_t ts_ns, const T& value) {
203 static_assert(sizeof(T) <= kPayloadSize, "Payload zu groß für Slot");

248// Die gesamte Cross-Language-Konstruktion steht und fällt damit, dass
249// std::atomic<T> byte-identisch zu T ist und ohne versteckten Lock
250// auskommt. Auf x86-64 mit gcc/clang ist beides gegeben, hier explizit
251// geprüft, damit ein Portierungsversuch auf eine krüppelige Plattform am
252// Compiler scheitert und nicht erst im Feld.
253static_assert(sizeof(std::atomic<std::uint32_t>) == sizeof(std::uint32_t),
254 "std::atomic<uint32_t> ist nicht byte-identisch zu uint32_t");
255static_assert(sizeof(std::atomic<std::uint64_t>) == sizeof(std::uint64_t),
256 "std::atomic<uint64_t> ist nicht byte-identisch zu uint64_t");
257static_assert(std::atomic<std::uint32_t>::is_always_lock_free,
258 "std::atomic<uint32_t> ist nicht lock-free -- Python könnte "
259 "den Wert dann nicht direkt aus dem Speicher lesen");
260static_assert(std::atomic<std::uint64_t>::is_always_lock_free,
261 "std::atomic<uint64_t> ist nicht lock-free -- Python könnte "
262 "den Wert dann nicht direkt aus dem Speicher lesen");
