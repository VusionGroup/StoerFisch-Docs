42# '<' erzwingt little-endian UND packed (kein Alignment-Padding). Das passt
43# zum C++-Struct, weil dort alle Felder 4-Byte-aligned sind und der Compiler
44# deshalb ebenfalls kein Padding einfügt. Bei CommandResponse steht das
45# uint64 auf Offset 16, also bereits 8-aligned -- ebenfalls paddingfrei.
46# Die static_asserts in command.hpp halten das fest, layout_test.py prüft es.
47# Layout von fisch::CommandRequest (40 Byte). Die letzten beiden Felder waren
48# früher reserved[8]; sie tragen jetzt sample_rate_hz und center_freq_hz (je
49# uint32). Die Struct-GRÖSSE bleibt 40 Byte -- nur die Bedeutung der letzten
50# 8 Byte hat sich geändert.
51REQUEST_FORMAT = "<IIIIIIfIII" # magic,schema,seq,kind,target,pattern,gain,policy,rate,freq
52RESPONSE_FORMAT = "<IIIIQ8s" # 32 Byte

130def build_request(
131 kind: CommandKind,
132 command_seq: int,
133 *,
134 target_channel: int = 0,
135 pattern_id: int = 0,
136 tx_gain_db: float = 0.0,
137 overflow_policy: OverflowPolicy = OverflowPolicy.DROP_NEWEST,
138 sample_rate_hz: int = 0,
139 center_freq_hz: int = 0,
140) -> bytes:
141 """Serialisiert ein Kommando in einen 40-Byte-Frame.
142
143 Nicht benötigte Felder werden mit 0 gefüllt; der Core ignoriert sie je
144 nach `kind`. Es findet hier BEWUSST keine Validierung statt -- die ist
145 Aufgabe des Cores (validateCommand), und der Test muss auch ungültige
146 Frames erzeugen können.
147
148 Args:
149 kind: Art des Kommandos.
150 command_seq: Streng monoton steigend. Taucht in
151 ActuatorState.applied_command_seq (#6) wieder auf, sobald das
152 Kommando tatsächlich wirkt.
153 target_channel: nur bei SET_TARGET ausgewertet.
154 pattern_id: nur bei SET_TARGET ausgewertet.
155 tx_gain_db: nur bei SET_TX_GAIN ausgewertet. ABSOLUT, nie relativ --
156 siehe Idempotenz-Regel in command.hpp.
157 overflow_policy: nur bei SET_OVERFLOW_POLICY ausgewertet.
158 sample_rate_hz: nur bei SET_SAMPLE_RATE ausgewertet, in Hz (2..20 MHz).
159 Loest einen HW-Retune mit kurzer RX-Unterbrechung aus.
160 center_freq_hz: nur bei SET_CENTER_FREQ ausgewertet, in Hz. Ebenfalls
161 HW-Retune mit kurzer Unterbrechung.
162
163 Returns:
164 40 Byte, direkt übertragbar.
165 """
166 return struct.pack(
167 REQUEST_FORMAT,
168 COMMAND_MAGIC,
169 COMMAND_SCHEMA_VERSION,
170 command_seq,
171 int(kind),
172 target_channel,
173 pattern_id,
174 tx_gain_db,
175 int(overflow_policy),
176 sample_rate_hz,
177 center_freq_hz,

181def parse_response(raw: bytes) -> CommandResponse:
182 """Liest einen 32-Byte-Antwortframe.
183
184 Raises:
185 ValueError: bei falscher Länge, falschem Magic oder falscher
186 Schema-Version -- in allen drei Fällen stimmt die Gegenstelle
187 nicht mit unserem Schema überein, Retry hilft nicht.
188 """
189 if len(raw) != RESPONSE_SIZE:
190 raise ValueError(
191 f"Antwortframe hat {len(raw)} Byte, erwartet {RESPONSE_SIZE}"
192 )
193
194 magic, schema_version, command_seq, status, generation, _reserved = struct.unpack(
195 RESPONSE_FORMAT, raw
196 )
197
198 if magic != COMMAND_MAGIC:
199 raise ValueError(f"Antwort hat falsches Magic: 0x{magic:08x}")
200 if schema_version != COMMAND_SCHEMA_VERSION:
201 raise ValueError(
202 f"Antwort hat Schema-Version {schema_version}, "
203 f"erwartet {COMMAND_SCHEMA_VERSION} -- Core und Python-Client "
204 f"stammen aus verschiedenen Ständen."
205 )
206
207 try:
208 status_enum = CommandStatus(status)
209 except ValueError:
210 raise ValueError(f"Antwort hat unbekannten Status-Code {status}") from None
211
212 return CommandResponse(
213 command_seq=command_seq,
214 status=status_enum,
215 generation=generation,
216 )
