133async def test_socket_roundtrip(tmp_path):
134 """Echter Unix-Socket: ein Mini-Server, der Requests parst und OK antwortet."""
135 sock_path = str(tmp_path / "cmd.sock")
136 generation = 5
137
138 async def handle(reader, writer):
139 raw = await reader.readexactly(cmd.REQUEST_SIZE)
140 parsed = cmd.parse_request(raw)
141 writer.write(_pack_response(int(parsed["command_seq"]), cmd.CommandStatus.OK, generation))
142 await writer.drain()
143
144 server = await asyncio.start_unix_server(handle, path=sock_path)
145 try:
146 transport = SocketCommandTransport(sock_path)
147 await transport.connect()
148 frame = cmd.build_request(cmd.CommandKind.PING, command_seq=11)
149 resp = await asyncio.wait_for(transport.request(frame), timeout=5)
150 assert resp.command_seq == 11
151 assert resp.status == cmd.CommandStatus.OK
152 assert transport.generation == generation
153 await transport.close()
154 finally:
155 # server.wait_closed() haengt auf CPython 3.12.3 -- hier nicht noetig,
156 # close() genuegt (betrifft nur diesen Test-Server, nicht die App).
157 server.close()
