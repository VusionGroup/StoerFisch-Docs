1"""Synthetischer Spektral-Strom fuer den Fake-Betrieb (ROD_FAKE=1).
2
3Erzeugt float32-dB-Frames wie der echte Ring: Rauschboden, zwei "BLE"-Traeger
4und ein wandernder Stoerer. Der Stoerer kann sich ueber ein Callable am
5Aktuator-Zustand der Fake-Telemetrie ausrichten, damit "Jammer an, Kanal N"
6auch im Wasserfall zu sehen ist.
7
8Genauso haengt die SDR-Sicht dran: nach einem Retune verschiebt sich das
9Fenster hier wie beim echten Core, und waehrend der RX-Unterbrechung kommt gar
10kein Frame. Jeder Frame traegt die Frequenz, mit der er "aufgenommen" wurde --
11auch im Fake darf sich niemand auf einen zwischengespeicherten Center-Wert
12verlassen.
13"""

70 def _bin_freqs(self) -> np.ndarray:

83

91 return None

95 t = time.monotonic() - self._t0
96 freqs = self._bin_freqs()

98 bins = -95.0 + self._rng.standard_normal(self._fft).astype(np.float32) * 1.5

100 def add_carrier(f0: float, amp_db: float, width_hz: float) -> None:
101 # Gauss-foermige Keule aufs Rauschband legen.
102 lobe = amp_db * np.exp(-0.5 * ((freqs - f0) / width_hz) ** 2)
103 np.maximum(bins, -95.0 + lobe, out=bins)
104
105 span = self._sample_rate * 0.35
106 add_carrier(self._center + span * np.sin(t * 0.7), 55.0,
107 self._sample_rate / self._fft * 3)
108 add_carrier(self._center - span * 0.5 + span * 0.3 * np.sin(t * 1.3 + 1.0),
109 45.0, self._sample_rate / self._fft * 2)
110
111 if self._jammer_probe is not None:
112 probe = self._jammer_probe()
113 if probe is not None:
114 enabled, channel, gain = probe
115 if enabled:
116 f0 = self._channel_freq(channel)
117 amp = 40.0 + gain
118 add_carrier(f0, amp, self._sample_rate / self._fft * 4)
119
120 self._seq += 1
121 self._frames_read += 1
122 return SpectrumFrame(
123 sequence=self._seq,
124 timestamp_ns=time.monotonic_ns(),
125 center_freq_hz=self._center,
126 bins=bins.astype(np.float32),
127 )
