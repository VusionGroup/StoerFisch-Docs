1"""Synthetischer Spektral-Strom fuer den Fake-Betrieb (ROD_FAKE=1).
2
3Erzeugt float32-dB-Frames wie der echte Ring: Rauschboden + zwei "BLE"-Traeger
4+ ein wandernder Stoerer. Optional richtet sich der Stoerer nach dem aktuellen
5Aktuator-Zustand der Fake-Telemetrie aus (gekoppelt ueber ein Callable), damit
6"Jammer an, Kanal N" auch im Wasserfall sichtbar wird.
7
8Ebenfalls gekoppelt ist die SDR-Sicht: Nach SET_SAMPLE_RATE / SET_CENTER_FREQ
9verschiebt sich das Fenster hier genauso wie beim echten Core, und waehrend der
10RX-Unterbrechung kommt gar kein Frame. Jeder Frame traegt die Frequenz, mit der
11er "aufgenommen" wurde -- Rod darf sich also auch im Fake nicht auf einen
12zwischengespeicherten Center-Wert verlassen.
13"""

94 def read_latest(self, max_drain: int) -> SpectrumFrame | None:
95 """Erzeugt den naechsten synthetischen Frame.
96
97 Folgt zuerst der SDR-Sonde (Retune uebernehmen); pausiert RX, kommt
98 ``None`` (kein Frame). Sonst: Rauschboden + zwei bewegte Traeger, und
99 falls der Jammer an ist, ein dritter Peak an der Aktuator-Frequenz.
100
101 Args:
102 max_drain: aus Protokollgruenden vorhanden; hier ohne Wirkung, weil
103 der Fake immer nur genau den aktuellen Frame erzeugt.
104 Returns:
105 Ein ``SpectrumFrame`` oder ``None``, wenn RX gerade pausiert.
106 """
107 if self._sdr_probe is not None:
108 center, sample_rate, rx_paused = self._sdr_probe()
109 if rx_paused:
110 # RX steht: kurze Retune-Luecke ODER laufende Sendung
111 # (Halbduplex). In beiden Faellen kommt kein Frame.
112 return None
113 self._center = center
114 self._sample_rate = sample_rate
115
116 t = time.monotonic() - self._t0
117 freqs = self._bin_freqs()
118
119 bins = -95.0 + self._rng.standard_normal(self._fft).astype(np.float32) * 1.5
120
121 def add_carrier(f0: float, amp_db: float, width_hz: float) -> None:
122 """Addiert eine gauss-foermige Traeger-Keule aufs Rauschband."""
123 lobe = amp_db * np.exp(-0.5 * ((freqs - f0) / width_hz) ** 2)
124 np.maximum(bins, -95.0 + lobe, out=bins)
125
126 span = self._sample_rate * 0.35
127 add_carrier(self._center + span * np.sin(t * 0.7), 55.0,
128 self._sample_rate / self._fft * 3)
129 add_carrier(self._center - span * 0.5 + span * 0.3 * np.sin(t * 1.3 + 1.0),
130 45.0, self._sample_rate / self._fft * 2)
131
132 if self._jammer_probe is not None:
133 probe = self._jammer_probe()
134 if probe is not None:
135 enabled, channel, gain = probe
136 if enabled:
137 f0 = self._channel_freq(channel)
138 amp = 40.0 + gain
139 add_carrier(f0, amp, self._sample_rate / self._fft * 4)
140
141 self._seq += 1
142 self._frames_read += 1
143 return SpectrumFrame(
144 sequence=self._seq,
145 timestamp_ns=time.monotonic_ns(),
146 center_freq_hz=self._center,
147 bins=bins.astype(np.float32),
148 )
