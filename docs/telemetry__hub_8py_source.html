1"""TelemetryHub: der EINE Poll-Task, der den Telemetrie-Ring liest und an alle
2WebSockets faechert.
3
4Warum zentral? Der Ring ist Single-Consumer (ein read_index) -- zwei parallele
5Leser wuerden sich Records klauen. Also pollt genau dieser Task, haelt je
6``kind`` den letzten Wert als Snapshot (damit ein frisch verbundener Browser
7sofort ein vollstaendiges Bild bekommt) und verteilt an pro-WebSocket-Queues.
8Laeuft eine Queue ueber (Browser zu langsam), wird ihr AELTESTER Eintrag
9verworfen -- Aktualitaet schlaegt Vollstaendigkeit im Live-Bild.
10
11Nebenbei zieht der Hub aus jedem ActuatorState (#6) die applied_command_seq in
12den PendingTracker und pollt den Dead-Man-Switch.
13"""

102 async def _run(self) -> None:
103 """Poll-Schleife: Ring drainen, Snapshot-Kinds coalescen und je Push-Tick
104 fanen, diskrete Events sofort fanen, Dead-Man pollen. Quellfehler werden
105 als ``source_error``-Event gemeldet statt die Schleife zu beenden."""
106 # Zwischen zwei Pushes gesammelte Snapshot-Kinds: pro Kind ueberlebt nur
107 # der zuletzt gesehene Wert (Coalescing). Diskrete Events werden NICHT
108 # hier gesammelt, sondern sofort gefanned (siehe unten).
109 pending_snapshot: dict[str, dict[str, Any]] = {}
110 push_accum = 0.0
111 while True:
112 try:
113 messages = self._source.poll()
114 except Exception as exc: # StaleSegment o.ae. -> als Event melden
115 self._fanout({
116 "kind": "source_error",
117 "timestamp_ns": time.monotonic_ns(),
118 "error": str(exc),
119 })
120 await asyncio.sleep(0.5)
121 continue
122
123 for msg in messages:
124 kind = msg.get("kind", "")
125 if kind == "actuator_state":
126 self._pending.update_applied(int(msg.get("applied_command_seq", 0)))
127 if kind in _SNAPSHOT_KINDS:
128 # Snapshot fuer Replay sofort aktuell halten, aber den Push
129 # bis zum naechsten Push-Tick zusammenfassen.
130 self._snapshot[kind] = msg
131 pending_snapshot[kind] = msg
132 else:
133 # Detection/Safety/Source-Error: selten und einzeln relevant
134 # (fuellen das Event-Log) -> ungedrosselt durchreichen.
135 self._fanout(msg)
136
137 self._last_dropped = self._source.dropped_count()
138
139 if self._safety.poll_deadman():
140 self._fanout({
141 "kind": "safety_event",
142 "timestamp_ns": time.monotonic_ns(),
143 "event": "deadman_tripped",
144 "message": "Dead-Man ausgeloest: kein Heartbeat, TX entschaerft.",
145 })
146 if self._on_deadman is not None:
147 asyncio.create_task(self._on_deadman()) # noqa: RUF006
148
149 push_accum += self._interval
150 if push_accum >= self._push_interval:
151 push_accum = 0.0
152 for msg in pending_snapshot.values():
153 self._fanout(msg)
154 pending_snapshot.clear()
155
156 await asyncio.sleep(self._interval)
157
158 def _fanout(self, message: dict[str, Any]) -> None:
159 """Legt die Nachricht in jede Abonnenten-Queue. Ist eine voll (Browser zu
160 langsam), wird ihr aeltester Eintrag verworfen -- Aktualitaet vor Vollstaendigkeit."""
161 for queue in self._subscribers:
162 if queue.full():
163 with contextlib.suppress(asyncio.QueueEmpty):
164 queue.get_nowait()
165 with contextlib.suppress(asyncio.QueueFull):
166 queue.put_nowait(message)
