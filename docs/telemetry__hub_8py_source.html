1"""Der eine Poll-Task, der den Telemetrie-Ring liest und an alle WebSockets verteilt.
2
3Zentral deshalb, weil der Ring Single-Consumer ist: es gibt nur einen
4read_index, zwei parallele Leser wuerden sich gegenseitig Records wegnehmen.
5Also pollt genau dieser Task, merkt sich je kind den letzten Wert als Snapshot,
6damit ein frisch verbundener Browser sofort ein vollstaendiges Bild bekommt,
7und schiebt alles in eine Queue pro WebSocket. Laeuft eine Queue ueber, weil
8ein Browser nicht hinterherkommt, fliegt ihr aeltester Eintrag raus -- im
9Live-Bild ist Aktualitaet mehr wert als Vollstaendigkeit.
10
11Nebenbei zieht der Hub die applied_command_seq aus jedem ActuatorState in den
12PendingTracker und pollt den Dead-Man-Switch.
13"""

104 async def _run(self) -> None:
105 # Zwischen zwei Pushes gesammelte Snapshot-Kinds. Pro Kind ueberlebt nur
106 # der zuletzt gesehene Wert; diskrete Events landen hier gar nicht,
107 # die gehen weiter unten sofort raus.
108 pending_snapshot: dict[str, dict[str, Any]] = {}
109 push_accum = 0.0
110 while True:
111 try:
112 messages = self._source.poll()
113 except Exception as exc: # StaleSegment und Konsorten melden wir als Event
114 self._fanout({
115 "kind": "source_error",
116 "timestamp_ns": time.monotonic_ns(),
117 "error": str(exc),
118 })
119 await asyncio.sleep(0.5)
120 continue
121
122 for msg in messages:
123 kind = msg.get("kind", "")
124 if kind == "actuator_state":
125 self._pending.update_applied(int(msg.get("applied_command_seq", 0)))
126 if kind in _SNAPSHOT_KINDS:
127 # Der Snapshot fuers Replay ist sofort aktuell, der Push
128 # wartet bis zum naechsten Tick.
129 self._snapshot[kind] = msg
130 pending_snapshot[kind] = msg
131 else:
132 # Detection, Safety und Source-Error sind selten und einzeln
133 # interessant, sie fuellen das Event-Log. Also ungedrosselt.
134 self._fanout(msg)
135
136 self._last_dropped = self._source.dropped_count()
137
138 # Lease pruefen: Verbindungs-Timeout und Anwesenheits-Rueckfrage,
139 # letztere nur wenn mehr als einer zuschaut. Befoerdert notfalls.
140 if self._lease is not None:
141 self._lease.poll(self.subscriber_countsubscriber_count)
142
143 if self._safety.poll_deadman():
144 self._fanout({
145 "kind": "safety_event",
146 "timestamp_ns": time.monotonic_ns(),
147 "event": "deadman_tripped",
148 "message": "Dead-Man ausgeloest: kein Heartbeat, TX entschaerft.",
149 })
150 if self._on_deadman is not None:
151 asyncio.create_task(self._on_deadman()) # noqa: RUF006
152
153 push_accum += self._interval
154 if push_accum >= self._push_interval:
155 push_accum = 0.0
156 for msg in pending_snapshot.values():
157 self._fanout(msg)
158 pending_snapshot.clear()
159
160 await asyncio.sleep(self._interval)

162 def _fanout(self, message: dict[str, Any]) -> None:
163 """Legt die Nachricht in jede Abonnenten-Queue.
164
165 Ist eine voll, weil der Browser nicht hinterherkommt, fliegt ihr
166 aeltester Eintrag raus.
167 """
168 for queue in self._subscribers:
169 if queue.full():
170 with contextlib.suppress(asyncio.QueueEmpty):
171 queue.get_nowait()
172 with contextlib.suppress(asyncio.QueueFull):
173 queue.put_nowait(message)
