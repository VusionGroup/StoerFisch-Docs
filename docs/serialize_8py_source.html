27def _occupancy(payload: np.void) -> dict[str, Any]:
28 # Die Bitmaske wird hier zu bool[40] aufgefaltet.
29 mask = int(payload["channel_occupied_mask"])
30 energy = [int(e) for e in payload["energy_dbm"]]
31 occupied = [(mask >> k) & 1 == 1 for k in range(40)]
32 return {"occupied": occupied, "energy_dbm": energy}

118def serialize_record(record: np.void) -> dict[str, Any]:
119 """Wandelt einen Telemetrie-Record in ein JSON-taugliches dict.
120
121 Unbekannte Typen kommen als kind="unknown_<N>" ohne Payload zurueck; meist
122 heisst das, der C++-Stand ist neuer als dieses Modul.
123 """
124 rtype = int(record["type"])
125 base: dict[str, Any] = {
126 "sequence": int(record["sequence"]),
127 "timestamp_ns": int(record["timestamp_ns"]),
128 }
129
130 entry = _PAYLOAD_SERIALIZERS.get(rtype)
131 if entry is None:
132 base["kind"] = f"unknown_{rtype}"
133 return base
134
135 decoder, serializer = entry
136 payload = decoder(record)
137 base["kind"] = KIND_BY_TYPE[rtype]
138 base.update(serializer(payload))
139 return base
