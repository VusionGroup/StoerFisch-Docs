203class TelemetryRingReader:
204 """Zero-Copy-Consumer für den von C++ geschriebenen TelemetryRing.
205
206 Öffnet ein bestehendes Shared-Memory-Segment (erstellt vom C++-Core)
207 und legt numpy-Views für Header und Slots darüber. Schreibt selbst
208 nichts außer dem read_index.
209 """
210
211 shm: shared_memory.SharedMemory
212 header: np.ndarray
213 slots: np.ndarray
214 generation: int
215 """Kennung der Core-Instanz, an der wir beim open() hingen."""
216 shm_path: Path
217 """Pfad im tmpfs, für den Stale-Check."""
218 inode: int
219 """Inode des Segments beim open(). Legt der Core ein frisches Segment
220 unter demselben Namen an, ändert sich der Inode -- daran erkennen wir,

223 @classmethod
224 def open(cls, shm_name: str, expected_capacity: int | None = None) -> "TelemetryRingReader":
225 """Öffnet ein bestehendes Segment.
226
227 Args:
228 shm_name: Name des Segments. Führende Slashes werden entfernt;
229 "vusion_telemetry" und "/vusion_telemetry" sind äquivalent
230 und treffen beide das C++-seitige shm_open("/vusion_telemetry").
231 expected_capacity: Optional. Wenn gesetzt, wird gegen die im
232 Header hinterlegte capacity_slots geprüft. Die tatsächliche
233 Slot-Zahl kommt IMMER aus dem Header -- sie wird nicht mehr
234 vom Aufrufer vorgegeben, damit C++- und Python-Seite nicht
235 über einen von Hand gepflegten Wert auseinanderlaufen können.
236
237 Raises:
238 RingNotReadyError: Segment fehlt/ist noch nicht initialisiert -> retry.
239 RingSchemaError: inkompatibles Schema -> Neubau nötig.
240 """
241 # Namensnormalisierung: multiprocessing.shared_memory setzt auf
242 # Linux selbst ein "/" davor. Wird der Name MIT Slash übergeben,
243 # entsteht "//name" -- das funktioniert zwar, tauchte aber als
244 # verwirrendes '//vusion_test_a' in den resource_tracker-Warnungen
245 # auf. Deshalb hier einmal sauber normalisieren.
246 normalized = shm_name.lstrip("/")
247
248 try:
249 # create=False ist entscheidend: Python ist Consumer, nicht Owner.
250 shm = shared_memory.SharedMemory(name=normalized, create=False)
251 except FileNotFoundError as exc:
252 raise RingNotReadyError(f"Segment '{normalized}' existiert nicht") from exc
253 except ValueError as exc:
254 # Tritt auf, wenn der Core zwar shm_open() gemacht hat, aber
255 # ftruncate() noch aussteht -- Segment hat dann Länge 0.
256 raise RingNotReadyError(f"Segment '{normalized}' ist noch leer") from exc
257
258 # Bekannter Stolperstein (siehe Architekturdiskussion): der
259 # multiprocessing.resource_tracker registriert JEDES geoeffnete
260 # SharedMemory-Objekt automatisch, unabhaengig von create=False,
261 # und versucht es beim Prozessende zu unlinken -- obwohl C++ hier
262 # Owner ist. Ab Python 3.13 gibt es dafuer den offiziellen
263 # `track=False`-Parameter; auf aelteren Versionen (wie hier) muss
264 # das Segment explizit aus dem Tracker ausgetragen werden.
265 try:
266 from multiprocessing import resource_tracker
267 resource_tracker.unregister(shm._name, "shared_memory") # noqa: SLF001
268 except Exception:
269 pass # z.B. bereits Python 3.13+ mit track=False, oder Plattform ohne resource_tracker
270
271 if shm.size < HEADER_DTYPE.itemsize:
272 shm.close()
273 raise RingNotReadyError(
274 f"Segment '{normalized}' ist kleiner als der Header "
275 f"({shm.size} < {HEADER_DTYPE.itemsize}) -- ftruncate noch nicht durch?"
276 )
277
278 header = np.ndarray((1,), dtype=HEADER_DTYPE, buffer=shm.buf)
279
280 # magic ist das Publikations-Flag: der Core schreibt es als letztes
281 # Feld mit Release-Store. Sehen wir es, sind alle anderen
282 # Header-Felder garantiert vollständig. Sehen wir es nicht, ist der
283 # Core schlicht noch nicht fertig -> retry, kein Fehler.
284 if int(header["magic"][0]) != RING_MAGIC:
285 shm.close()
286 raise RingNotReadyError(
287 f"Segment '{normalized}': magic noch nicht gesetzt "
288 f"(Core initialisiert gerade?)"
289 )
290
291 if int(header["schema_version"][0]) != SCHEMA_VERSION:
292 found = int(header["schema_version"][0])
293 shm.close()
294 raise RingSchemaError(
295 f"Schema-Version mismatch für '{normalized}': "
296 f"erwartet {SCHEMA_VERSION}, gefunden {found}. "
297 f"C++-Core und Python-Layout stammen aus verschiedenen Ständen."
298 )
299
300 capacity = int(header["capacity_slots"][0])
301 if expected_capacity is not None and capacity != expected_capacity:
302 shm.close()
303 raise RingSchemaError(
304 f"capacity_slots mismatch für '{normalized}': "
305 f"Header sagt {capacity}, erwartet wurde {expected_capacity}"
306 )
307
308 needed = HEADER_DTYPE.itemsize + capacity * RECORD_DTYPE.itemsize
309 if shm.size < needed:
310 shm.close()
311 raise RingSchemaError(
312 f"Segment '{normalized}' zu klein: {shm.size} Byte, "
313 f"aber Header deklariert {capacity} Slots ({needed} Byte nötig)"
314 )
315
316 slots = np.ndarray(
317 (capacity,),
318 dtype=RECORD_DTYPE,
319 buffer=shm.buf,
320 offset=HEADER_DTYPE.itemsize,
321 )
322
323 path = _shm_path(normalized)
324 return cls(
325 shm=shm,
326 header=header,
327 slots=slots,
328 generation=int(header["generation"][0]),
329 shm_path=path,
330 inode=path.stat().st_ino,
331 )

333 def check_stale(self) -> None:
334 """Prüft, ob unser Mapping noch das aktuelle Segment ist.
335
336 Hintergrund: Startet der Core neu, unlinkt er den Namen und legt ein
337 FRISCHES Segment an. Unser Mapping bleibt gültig, zeigt aber auf die
338 alten Pages -- die sich nie wieder ändern. Ein poll() darauf liefert
339 stillschweigend für immer nichts. Weil die alten Pages auch die alte
340 `generation` behalten, nützt ein Vergleich der Generation im eigenen
341 Mapping hier NICHTS -- geprüft werden muss gegen den NAMEN im tmpfs.
342 Deshalb der Inode-Vergleich.
343
344 Dies ist die Notlösung, solange es keinen CommandChannel gibt. Sobald
345 der Socket steht (Phase 2), erkennt Python den Core-Tod direkt am
346 EOF der Verbindung; dieser Check bleibt dann als billiger Zusatz.
347
348 Raises:
349 StaleSegmentError: Segment wurde ersetzt oder entfernt -> der
350 Aufrufer muss close() und neu open() machen.
351 """
352 try:
353 current_inode = self.shm_path.stat().st_ino
354 except FileNotFoundError as exc:
355 raise StaleSegmentError(
356 f"Segment '{self.shm_path}' existiert nicht mehr -- Core beendet?"
357 ) from exc
358 if current_inode != self.inode:
359 raise StaleSegmentError(
360 f"Segment '{self.shm_path}' wurde ersetzt (Core-Neustart). "
361 f"Unser Mapping ist eingefroren -- neu öffnen."
362 )

364 def poll(self, check_liveness: bool = False) -> list[np.void]:
365 """Liest alle seit dem letzten poll() neu geschriebenen Records.
366
367 Entspricht dem Consumer-Ablauf aus der Architekturbesprechung:
368 write_index (acquire-artig) lesen, neue Slots kopieren, read_index
369 vorrücken. numpy/Python garantiert keine echte acquire/release-
370 Semantik -- für den nicht-zeitkritischen Analyse-Pfad mit
371 Poll-Intervall (z.B. alle 10 ms) ist das akzeptiert und
372 dokumentiert, siehe Diskussion zu Cross-Language-Atomics.
373
374 Args:
375 check_liveness: Wenn True, wird vorab check_stale() ausgeführt.
376 Das kostet einen stat()-Syscall -- daher nicht bei jedem
377 poll() sinnvoll, sondern z.B. jeden n-ten, oder immer dann,
378 wenn längere Zeit keine Daten kamen.
379
380 Returns:
381 Liste neuer Records (numpy structured scalars), älteste zuerst.
382
383 Raises:
384 StaleSegmentError: nur wenn check_liveness=True.
385 """
386 if check_liveness:
387 self.check_stale()
388
389 write = int(self.headerheader["write_index"][0])
390 read = int(self.headerheader["read_index"][0])
391 capacity = int(self.headerheader["capacity_slots"][0])
392
393 new_records: list[np.void] = []
394 idx = read
395 while idx != write:
396 new_records.append(self.slotsslots[idx].copy())
397 idx = (idx + 1) % capacity
398
399 self.headerheader["read_index"][0] = write
400 return new_records

416def _decode_payload(record: np.void, dtype: np.dtype) -> np.void:
417 """Interpretiert die ersten `dtype.itemsize` Bytes des 64-Byte-Payloads
418 als das angegebene Struct.
419
420 `count=1` ist hier zwingend und nicht kosmetisch: np.frombuffer() ohne
421 count verlangt, dass die Puffergröße ein exaktes Vielfaches der
422 Elementgröße ist. Der Payload ist immer 64 Byte gross, also würde das
423 nur für dtypes funktionieren, deren Größe 64 teilt (z.B.
424 LatencyTimestamps mit 32) und für alle anderen mit
425 "buffer size must be a multiple of element size" fehlschlagen. Mit
426 count=1 wird genau ein Element gelesen und der ungenutzte Rest des
427 Payloads ignoriert -- was genau der Absicht entspricht, da setPayload()
428 auf der C++-Seite den Rest mit Nullen füllt.
429 """
430 return np.frombuffer(record["payload"].tobytes(), dtype=dtype, count=1)[0]
