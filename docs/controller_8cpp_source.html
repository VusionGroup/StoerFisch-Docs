31int Controller::selectTarget(std::uint64_t occupancy_mask,
32 const std::vector<float>& channel_energies) {
33 int best = -1;
34 float best_energy = 0.0f;
35 for (std::uint32_t k = 0; k < kBleChannels; ++k) {
36 if ((occupancy_mask >> k) & 1u) {
37 const float e = channel_energies[k];
38 if (best < 0 || e > best_energy) {
39 best = static_cast<int>(k);
40 best_energy = e;
41 }
42 }
43 }
44 return best;
45}

47ControllerOutput Controller::decide(std::uint64_t occupancy_mask,
48 const std::vector<float>& channel_energies) {
49 ControllerOutput out{};
50
51 // Nicht aktiv -> Idle, Jammer aus. Kurzschluss, bevor irgendetwas
52 // ausgewertet wird.
53 if (!running_ || channel_energies.size() < kBleChannels) {
54 state_ = running_ ? state_ : ControllerState::Idle;
55 out.state = state_;
56 out.decision = ControllerDecision::Disable;
57 out.target_channel = target_channel_;
58 out.jammer_enabled = false;
59 out.error_metric = 0.0f;
60 return out;
61 }
62
63 const int candidate = selectTarget(occupancy_mask, channel_energies);
64
65 // Zustandsmaschine.
66 if (candidate < 0) {
67 // Kein belegter Kanal sichtbar.
68 if (has_target_) {
69 // Ziel gerade verloren -> in Lost gehen und Haltezeit starten.
70 if (state_ != ControllerState::Lost) {
71 state_ = ControllerState::Lost;
72 lost_counter_ = 0;
73 } else {
74 ++lost_counter_;
75 if (lost_counter_ >= config_.lost_hold_cycles) {
76 // Haltezeit abgelaufen -> Ziel endgültig aufgeben.
77 has_target_ = false;
78 state_ = ControllerState::Searching;
79 }
80 }
81 } else {
82 state_ = ControllerState::Searching;
83 }
84
85 // In Lost halten wir den Jammer noch auf dem alten Ziel (Annahme: das
86 // BLE-Signal kommt beim nächsten Hop womöglich zurück). In Searching
87 // ist er aus.
88 const bool still_holding = (state_ == ControllerState::Lost);
89 out.state = state_;
90 out.decision = still_holding ? ControllerDecision::Hold : ControllerDecision::Disable;
91 out.target_channel = target_channel_;
92 out.jammer_enabled = still_holding && has_target_;
93 out.error_metric = still_holding ? 1.0f : 0.0f; // "Ziel unsichtbar"
94 return out;
95 }
96
97 // Es gibt einen belegten Kanal.
98 const auto new_target = static_cast<std::uint32_t>(candidate);
99
100 if (!has_target_) {
101 // Erstes Ziel gefunden.
102 target_channel_ = new_target;
103 has_target_ = true;
104 lost_counter_ = 0;
105 state_ = ControllerState::Tracking;
106 out.state = state_;
107 out.decision = ControllerDecision::Enable;
108 out.target_channel = target_channel_;
109 out.jammer_enabled = true;
110 out.error_metric = 0.0f;
111 return out;
112 }
113
114 // Wir hatten bereits ein Ziel.
115 lost_counter_ = 0;
116 if (new_target == target_channel_) {
117 // Ziel unverändert -> halten.
118 state_ = ControllerState::Tracking;
119 out.state = state_;
120 out.decision = ControllerDecision::Hold;
121 out.target_channel = target_channel_;
122 out.jammer_enabled = true;
123 out.error_metric = 0.0f;
124 return out;
125 }
126
127 // Ziel hat gewechselt (BLE ist gehoppt) -> nachführen.
128 // error_metric = Distanz des Hops in Kanälen, als grobes Maß der
129 // Nachführ-"Anstrengung". Nützt später der Reglercharakterisierung (#5).
130 const float hop_distance = static_cast<float>(
131 new_target > target_channel_ ? new_target - target_channel_
132 : target_channel_ - new_target);
133 target_channel_ = new_target;
134 state_ = ControllerState::Tracking;
135 out.state = state_;
136 out.decision = ControllerDecision::Retune;
137 out.target_channel = target_channel_;
138 out.jammer_enabled = true;
139 out.error_metric = hop_distance;
140 return out;
141}
