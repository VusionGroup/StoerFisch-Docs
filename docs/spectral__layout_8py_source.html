67def make_slot_dtype(fft_size: int, record_stride: int) -> np.dtype:
68 """Baut das vollständige Slot-dtype zur Laufzeit.
69
70 Der Slot besteht aus dem festen Header, den fft_size floats und einem
71 Padding-Rest bis record_stride (der Producer rundet den Stride auf ein
72 Vielfaches von 8 auf). Das Padding wird als Bytes mitgeführt, damit das
73 dtype exakt record_stride groß ist und aufeinanderfolgende Slots korrekt
74 aneinandergrenzen.
75
76 Args:
77 fft_size: Bins pro Frame, aus dem Header.
78 record_stride: Bytes pro Slot, aus dem Header.
79
80 Raises:
81 ValueError: wenn Header und fft_size nicht zusammenpassen.
82 """
83 fixed = SPECTRAL_SLOT_HEADER_DTYPE.itemsize + fft_size * np.dtype("<f4").itemsize
84 if record_stride < fixed:
85 raise ValueError(
86 f"record_stride ({record_stride}) kleiner als Header+Bins ({fixed}) "
87 f"-- Layout inkonsistent"
88 )
89 pad = record_stride - fixed
90 fields = [
91 ("seq", "<u4"),
92 ("sequence", "<u4"),
93 ("timestamp_ns", "<u8"),
94 ("center_freq_hz", "<f8"),
95 ("bins", "<f4", (fft_size,)),
96 ]
97 if pad:
98 fields.append(("_pad", "u1", (pad,)))
99 return np.dtype(fields)

120class SpectralRingReader:
121 """Zero-Copy-fähiger Consumer des Spektral-Rings.
122
123 "Zero-Copy-fähig" mit Einschränkung: Der numpy-View auf den Ring ist
124 zero-copy, aber read_frame() macht bewusst eine KOPIE der Bins, weil das
125 Seqlock-Protokoll erst NACH dem Lesen feststellen kann, ob die Daten
126 konsistent waren. Einen View direkt weiterzugeben wäre unsicher -- der
127 Producer könnte ihn jederzeit überschreiben. Für den nicht-zeitkritischen
128 Analyse-Pfad ist die Kopie akzeptabel (eine 4-KB-FFT zu kopieren ist
129 vernachlässigbar gegenüber ihrer Verarbeitung).
130 """

132 def __init__(self, shm_name: str, from_start: bool = False):
133 """Öffnet ein bestehendes Segment.
134
135 Args:
136 shm_name: Name des Segments (führende Slashes werden entfernt).
137 from_start: Wenn False (Standard, Live-Betrieb), startet der Reader
138 an der aktuellen Schreibposition und sieht nur künftige Frames.
139 Wenn True, startet er bei Index 0 und liest auch bereits
140 gepufferte Frames -- gedacht für Offline-Auswertung eines
141 gefüllten Segments und für Tests. Achtung: Bei True kann der
142 Reader sofort überrundet sein, wenn der Producer den Ring
143 bereits mehr als capacity-mal beschrieben hat; die
144 Überrundungslogik in read_frame() fängt das ab.

233 def read_frame(self) -> SpectralFrame | None:
234 """Liest den nächsten konsistenten Frame. Blockiert nie.
235
236 Implementiert dasselbe Protokoll wie SpectralRingReader::tryPop() in
237 C++: Überrundung erkennen und vorspringen, dann Seqlock (seq gerade,
238 kopieren, seq erneut prüfen). Bei Torn Read: Frame verwerfen, zählen,
239 None zurückgeben (nächster Aufruf holt den nächsten Slot).
240
241 Returns:
242 SpectralFrame bei Erfolg, None wenn leer ODER gerade ein Torn/Lap
243 passiert ist. Der Aufrufer ruft in einer Poll-Schleife wiederholt
244 auf; None heißt schlicht "jetzt gerade nichts Gültiges".
245 """
246 write = int(self._header["write_index"][0])
247 if self._read_pos >= write:
248 return None # leer
249
250 # Überrundung: Producer ist um mehr als capacity vorausgeeilt.
251 if write - self._read_pos > self._capacity:
252 self._lapped += (write - self._read_pos) - self._capacity
253 self._read_pos = write - self._capacity
254
255 slot = self._slots[self._read_pos % self._capacity]
256
257 # --- Seqlock-Lesen ---
258 seq1 = int(slot["seq"])
259 if seq1 & 1:
260 # Producer schreibt genau diesen Slot gerade -> verloren.
261 self._torn += 1
262 self._read_pos += 1
263 self._header["read_index"][0] = self._read_pos
264 return None
265
266 # Kopie ziehen, solange seq gerade ist.
267 sequence = int(slot["sequence"])
268 timestamp_ns = int(slot["timestamp_ns"])
269 center_freq_hz = float(slot["center_freq_hz"])
270 bins = np.array(slot["bins"], dtype=np.float32) # explizite Kopie
271
272 seq2 = int(slot["seq"])
273 self._read_pos += 1
274 self._header["read_index"][0] = self._read_pos
275
276 if seq1 != seq2:
277 # Während des Kopierens überschrieben -> Mischmasch, verwerfen.
278 self._torn += 1
279 return None
280
281 return SpectralFrame(
282 sequence=sequence,
283 timestamp_ns=timestamp_ns,
284 center_freq_hz=center_freq_hz,
285 bins=bins,
286 )
