11CommandStatus validateCommand(const CommandRequest& req) noexcept {
12 // Zuerst das Schema: Passt das nicht, sind alle weiteren Felder
13 // bedeutungslos, weil wir sie womöglich an der falschen Stelle lesen.
14 if (req.magic != kCommandMagic) {
15 return CommandStatus::SchemaMismatch;
16 }
17 if (req.schema_version != kCommandSchemaVersion) {
18 return CommandStatus::SchemaMismatch;
19 }
20
21 switch (static_cast<CommandKind>(req.kind)) {
22 case CommandKind::Ping:
23 case CommandKind::Start:
24 case CommandKind::Stop:
25 case CommandKind::Shutdown:
26 // Werten keine Parameterfelder aus -- nichts zustandslos zu prüfen.
27 // (Start/Stop können sehr wohl scheitern, aber erst am
28 // Controller-Zustand; das ist nicht Aufgabe dieser Funktion.)
29 return CommandStatus::Ok;
30
31 case CommandKind::SetTarget:
32 if (req.target_channel > kMaxBleChannel) {
33 return CommandStatus::InvalidChannel;
34 }
35 // pattern_id bewusst NICHT geprüft -- siehe command.hpp.
36 return CommandStatus::Ok;
37
38 case CommandKind::SetTxGain:
39 // Positiv formuliert statt "< min || > max": Jeder Vergleich mit
40 // NaN liefert false, also fällt NaN hier automatisch durch. Bei
41 // einem Wert, der aus einer Python-Berechnung stammt, ist NaN ein
42 // realistischer Unfall -- und ein NaN an den TX-Gain zu geben wäre
43 // ein stiller, sehr unangenehmer Fehler.
44 if (!(req.tx_gain_db >= kMinTxGainDb && req.tx_gain_db <= kMaxTxGainDb)) {
45 return CommandStatus::InvalidGain;
46 }
47 return CommandStatus::Ok;
48
49 case CommandKind::SetOverflowPolicy:
50 if (req.overflow_policy > kMaxOverflowPolicyValue) {
51 return CommandStatus::InvalidPolicy;
52 }
53 return CommandStatus::Ok;
54
55 case CommandKind::SetSampleRate:
56 if (req.sample_rate_hz < kMinSampleRateHz ||
57 req.sample_rate_hz > kMaxSampleRateHz) {
58 return CommandStatus::InvalidSampleRate;
59 }
60 return CommandStatus::Ok;
61
62 case CommandKind::SetCenterFreq:
63 if (req.center_freq_hz < kMinCenterFreqHz ||
64 req.center_freq_hz > kMaxCenterFreqHz) {
65 return CommandStatus::InvalidFrequency;
66 }
67 return CommandStatus::Ok;
68
69 case CommandKind::StartRecording:
70 case CommandKind::StopRecording:
71 // Parameterlos -- zustandslos immer gültig. Ob eine Aufnahme
72 // tatsächlich startet (Verzeichnis schreibbar?) entscheidet sich
73 // erst beim Ausführen; das ist ein Laufzeit-, kein Validierungsfehler.
74 return CommandStatus::Ok;
75
76 case CommandKind::StartJamming:
77 // Braucht einen gültigen Zielkanal und ein bekanntes Störmuster
78 // (0=Dauerträger, 1=Rauschen).
79 if (req.target_channel > kMaxBleChannel) {
80 return CommandStatus::InvalidChannel;
81 }
82 if (req.pattern_id > 1) {
83 return CommandStatus::InvalidPattern;
84 }
85 return CommandStatus::Ok;
86
87 case CommandKind::StartReplay:
88 // target_channel darf 0xFFFFFFFF sein (= auf Originalfrequenz
89 // abspielen) oder ein gültiger BLE-Kanal (= dorthin verschieben).
90 if (req.target_channel != 0xFFFFFFFFu &&
91 req.target_channel > kMaxBleChannel) {
92 return CommandStatus::InvalidChannel;
93 }
94 return CommandStatus::Ok;
95
96 case CommandKind::StopTransmit:
97 return CommandStatus::Ok;
98 }
99
100 // Kein bekannter Enumerator getroffen. Erreichbar, weil `kind` als roher
101 // uint32 über die Leitung kommt und beliebige Werte enthalten kann.
102 return CommandStatus::UnknownKind;
103}
