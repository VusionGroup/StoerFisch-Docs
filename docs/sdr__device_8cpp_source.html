59SdrDevice::SdrDevice(const std::string& serial) {
60 // Library-Referenz zuerst: hält libhackrf am Leben, solange irgendein
61 // SdrDevice existiert. Wirft bei Fehler, dann passiert unten nichts mehr.
62 acquireLibrary();
63 holds_library_ref_ = true;
64
65 int result;
66 if (serial.empty()) {
67 result = hackrf_open(&device_);
68 } else {
69 result = hackrf_open_by_serial(serial.c_str(), &device_);
70 }
71 if (result != HACKRF_SUCCESS) {
72 // Der Destruktor läuft bei einem geworfenen Konstruktor NICHT --
73 // die Library-Referenz müssen wir hier von Hand zurückgeben, sonst
74 // bleibt libhackrf für immer initialisiert.
75 releaseLibrary();
76 holds_library_ref_ = false;
77 throw std::runtime_error(
78 std::string("hackrf_open") + (serial.empty() ? "" : "_by_serial('" + serial + "')") +
79 " fehlgeschlagen: " + hackrf_error_name(static_cast<hackrf_error>(result)) +
80 (serial.empty()
81 ? " -- ist ein HackRF angesteckt? (hackrf_info pruefen)"
82 : " -- gibt es ein Geraet mit dieser Seriennummer? (hackrf_info pruefen)"));
83 }
84 serial_ = serial;
85}

87SdrDevice::~SdrDevice() {
88 // Reihenfolge ist wichtig: erst RX stoppen, dann Gerät schließen, dann
89 // (ggf.) die Library. Die Library-Referenz geht als letztes weg, weil
90 // library_ref_ nach device_ deklariert ist -- der Destruktor läuft in
91 // umgekehrter Deklarationsreihenfolge. Hier trotzdem explizit, damit die
92 // Absicht sichtbar ist.
93 if (streaming_ && device_ != nullptr) {
94 hackrf_stop_rx(device_); // Rückgabe im Destruktor bewusst ignoriert
95 streaming_ = false;
96 }
97 if (device_ != nullptr) {
98 hackrf_close(device_);
99 device_ = nullptr;
100 }
101 if (holds_library_ref_) {
102 releaseLibrary(); // hackrf_exit() nur, wenn wir das letzte Gerät waren
103 holds_library_ref_ = false;
104 }
105}

107void SdrDevice::acquireLibrary() {
108 std::lock_guard<std::mutex> lock(libraryMutex());
109 if (libraryRefcount() == 0) {
110 const int result = hackrf_init();
111 if (result != HACKRF_SUCCESS) {
112 throw std::runtime_error(
113 std::string("hackrf_init fehlgeschlagen: ") +
114 hackrf_error_name(static_cast<hackrf_error>(result)));
115 }
116 }
117 ++libraryRefcount();
118}

120void SdrDevice::releaseLibrary() noexcept {
121 std::lock_guard<std::mutex> lock(libraryMutex());
122 if (libraryRefcount() > 0 && --libraryRefcount() == 0) {
123 hackrf_exit();
124 }

140void SdrDevice::setRxGain(std::uint32_t lna_gain_db, std::uint32_t vga_gain_db) {
141 // Auf das Hardware-Raster abrunden statt einen Fehler zu werfen: die
142 // Hardware ignoriert ohnehin nicht-rasternde Werte bzw. rundet selbst;
143 // wir machen es explizit und vorhersagbar.
144 if (lna_gain_db > kLnaMaxDb) lna_gain_db = kLnaMaxDb;
145 if (vga_gain_db > kVgaMaxDb) vga_gain_db = kVgaMaxDb;
146 lna_gain_db -= (lna_gain_db % 8); // 8-dB-Raster
147 vga_gain_db -= (vga_gain_db % 2); // 2-dB-Raster
148
149 checkResult(hackrf_set_lna_gain(device_, lna_gain_db), "hackrf_set_lna_gain");
150 checkResult(hackrf_set_vga_gain(device_, vga_gain_db), "hackrf_set_vga_gain");
151}

230int SdrDevice::fillTx(hackrf_transfer* transfer) {
231 if (!tx_callback_) {
232 // Kein Callback -> nichts senden (Nullen = Trägerpause).
233 std::memset(transfer->buffer, 0, static_cast<std::size_t>(transfer->buffer_length));
234 transfer->valid_length = transfer->buffer_length;
235 return 0;
236 }
237
238 auto* buf = reinterpret_cast<std::int8_t*>(transfer->buffer);
239 const std::size_t capacity = static_cast<std::size_t>(transfer->buffer_length);
240
241 const std::size_t filled = tx_callback_(buf, capacity);
242
243 // Rest des Puffers mit Null auffüllen, falls der Callback weniger lieferte
244 // (sonst würde alter Pufferinhalt gesendet -> Störspitzen).
245 if (filled < capacity) {
246 std::memset(buf + filled, 0, capacity - filled);
247 }
248 transfer->valid_length = transfer->buffer_length;
249
250 // Callback lieferte 0 Bytes -> keine Daten mehr -> Stream beenden lassen.
251 // libhackrf ruft danach den flush-Callback und stoppt. valid_length bleibt
252 // voll (mit Nullen), damit der letzte Transfer sauber durchläuft.
253 return (filled == 0) ? 1 : 0;
254}

256void SdrDevice::dispatch(const hackrf_transfer* transfer) {
257 if (!rx_callback_) {
258 return;
259 }
260
261 // valid_length ist die Zahl tatsächlich übertragener Bytes (kann kleiner
262 // als buffer_length sein). Zwei Bytes = ein komplexes Sample (I,Q).
263 const int byte_count = transfer->valid_length;
264 if (byte_count <= 0) {
265 return;
266 }
267 const std::size_t num_samples = static_cast<std::size_t>(byte_count) / 2;
268
269 // Umwandlungspuffer wiederverwenden, um Allokation im Streaming-Pfad zu
270 // vermeiden (der Callback ist zeitkritisch).
271 iq_scratch_.resize(num_samples * 2);
272
273 const auto* raw = reinterpret_cast<const std::int8_t*>(transfer->buffer);
274 constexpr float kScale = 1.0f / 128.0f; // int8 [-128,127] -> ~[-1, 1)
275 for (std::size_t i = 0; i < num_samples * 2; ++i) {
276 iq_scratch_[i] = static_cast<float>(raw[i]) * kScale;
277 }
278
279 IqBlock block{iq_scratch_.data(), num_samples, raw};
280 rx_callback_(block);
281}
