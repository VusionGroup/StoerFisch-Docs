59async def test_retune_defers_applied_command_seq(monkeypatch):
60 tel = FakeTelemetrySource()
61 transport = FakeCommandTransport(tel)
62 ctrl = _controller(transport)
63
64 now = [1000.0]
65 monkeypatch.setattr("rod.sources.fake_telemetry.time.monotonic", lambda: now[0])
66
67 await ctrl.set_center_freq(2_426_000_000)
68
69 # Waehrend der Unterbrechung: keine Records, Ebene 3 noch nicht erreicht.
70 now[0] += 0.05
71 assert tel.poll() == []
72 assert tel.sdr_state()[2] is True
73
74 # Nach der Unterbrechung zieht applied_command_seq nach.
75 now[0] += 0.2
76 records = tel.poll()
77 assert records
78 act = next(r for r in records if r["kind"] == "actuator_state")
79 assert act["applied_command_seq"] == 1

81 assert tel.sdr_state()[2] is False
