1"""Ein echter TelemetryRing in Shared Memory, produzentenseitig -- ohne C++.
2
3Damit koennen die Reader/Sources gegen ein byte-echtes Segment getestet werden.
4Konvention wie im C++-Core: write_index/read_index sind GEWRAPPTE Indizes
5(0..capacity-1), ein Slot bleibt frei (voll = (write+1)%cap == read), und der
6Ring ist drop-newest (bei voll wird dropped_count erhoeht, nichts ueberschrieben).
7Der Producer ist Owner -> er unlinkt beim close().
8"""

33 self.header["read_index"][0] = 0

39 def _push(self, rtype: int, payload: np.ndarray) -> None:
40 w = int(self.header["write_index"][0])
41 r = int(self.header["read_index"][0])
42 if (w + 1) % self.capacity == r: # voll -> drop newest
43 self.header["dropped_count"][0] += 1
44 return
45 self._seq += 1
46 slot = self.slots[w]
47 slot["timestamp_ns"] = self._seq * 1000
48 slot["type"] = rtype
49 slot["sequence"] = self._seq
50 raw = payload.tobytes()
51 buf = np.zeros(tel.PAYLOAD_SIZE, dtype="u1")
52 buf[: len(raw)] = np.frombuffer(raw, dtype="u1")
53 slot["payload"] = buf
54 self.header["write_index"][0] = (w + 1) % self.capacity
