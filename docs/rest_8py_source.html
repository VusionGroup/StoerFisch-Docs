121def _map_command_error(exc: Exception) -> HTTPException:
122 if isinstance(exc, TxSafetyError):
123 return HTTPException(status_code=409, detail=f"TX gesperrt: {exc}")
124 if isinstance(exc, cmd.CommandError):
125 detail = f"Kommando abgelehnt: {exc.status.name}"
126 if exc.status == cmd.CommandStatus.UNKNOWN_KIND:
127 detail = f"{detail} -- {_UNKNOWN_KIND_HINT}"
128 return HTTPException(status_code=409, detail=detail)
129 if isinstance(exc, CommandTransportError):
130 return HTTPException(status_code=503, detail=f"Kommandopfad tot: {exc}")
131 return HTTPException(status_code=500, detail=str(exc))

144def require_writer(request: Request, x_owner: str | None = Header(default=None)) -> str:
145 """Laesst nur den aktuellen Schreibrecht-Inhaber durch.
146
147 Die owner-id kommt aus dem X-Owner-Header und wird mit dem Writer der Lease
148 verglichen. Zuschauer bekommen 403; damit ist jede schreibende Route
149 serverseitig gesperrt und nicht nur im UI ausgegraut. Zurueck kommt die
150 owner-id, damit die Route sie nicht noch einmal auslesen muss.
151 """
152 lease = request.app.state.lease
153 if not lease.is_writer(x_owner):
154 raise HTTPException(
155 status_code=403,
156 detail="Kein Schreibrecht. Erst ueber POST /api/control/request "
157 "anfordern (und per Telemetrie-Heartbeat halten).",
158 )
159 # Ein echtes Steuerkommando ist der beste Anwesenheitsbeweis: es setzt die
160 # Untaetigkeitsuhr zurueck und beendet eine laufende Rueckfrage.
161 lease.confirm_presence(x_owner)
162 return x_owner # type: ignore[return-value] # is_writer schliesst None aus
