81# UNKNOWN_KIND heisst nicht "kaputtes Kommando", sondern "der Core hat diesen
82# CommandKind nicht erkannt". Der Handshake kann das nicht abfangen:
83# COMMAND_SCHEMA_VERSION ist bei der Einfuehrung von SET_SAMPLE_RATE/
84# SET_CENTER_FREQ auf 1 geblieben.
85#
86# BEKANNTER CORE-BEFUND (Stand 27.07.2026): command.hpp deklariert Kind 7/8 samt

88# den Retune vollstaendig -- aber validateCommand() in src/command/command.cpp
89# hat keine case-Zweige dafuer. Kommandos fallen dort auf UnknownKind durch und
90# werden verworfen, BEVOR sie die Queue erreichen. Das ist ein Core-Fix (zwei
91# case-Zweige), kein Rod-Problem und kein veraltetes Binary.
92_UNKNOWN_KIND_HINT = (
93 "Der Core hat dieses Kommando nicht erkannt (UNKNOWN_KIND). Bei den neueren "
94 "Kommandos (Retune 7/8, Recording 9/10, Jamming/Replay 11-13) ist das im "
95 "Core-Stand vom 27.07.2026 zu erwarten: validateCommand() in "
96 "src/command/command.cpp deckt sie nicht ab, die Kommandos werden vor der "
97 "Queue verworfen. Fix gehoert in den Core (siehe Task.md)."

101def _map_command_error(exc: Exception) -> HTTPException:
102 """Bildet Kommandopfad-Fehler auf HTTP ab: CommandError->409, Transport tot->503."""
103 if isinstance(exc, TxSafetyError):
104 return HTTPException(status_code=409, detail=f"TX gesperrt: {exc}")
105 if isinstance(exc, cmd.CommandError):
106 detail = f"Kommando abgelehnt: {exc.status.name}"
107 if exc.status == cmd.CommandStatus.UNKNOWN_KIND:
108 detail = f"{detail} -- {_UNKNOWN_KIND_HINT}"
109 return HTTPException(status_code=409, detail=detail)
110 if isinstance(exc, CommandTransportError):
111 return HTTPException(status_code=503, detail=f"Kommandopfad tot: {exc}")
112 return HTTPException(status_code=500, detail=str(exc))

158@router.post("/sdr")
159async def set_sdr(body: SdrBody, request: Request) -> dict[str, Any]:
160 """Stellt die SDR-Hardware im Core um (Retune).
161
162 Anders als frueher ist das ein echtes Kommando: der Core stoppt den
163 RX-Stream, stimmt den HackRF um und startet neu. Fuer einige Millisekunden
164 kommen weder Spektral-Frames noch Telemetrie -- das ist erwartet. Ack-Ebene
165 3 (``applied_seq`` in /api/status) zieht entsprechend spaeter nach; erst
166 dann gehoeren die Frames zur neuen Einstellung.
167 """
168 controller = request.app.state.controller
169 try:
170 responses = await controller.set_sdr(
171 SdrConfig(sample_rate_hz=body.sample_rate_hz,
172 center_freq_hz=body.center_freq_hz)
173 )
174 except Exception as exc:
175 raise _map_command_error(exc) from exc
176 return {
177 "ok": True,
178 "sdr": controller.sdr.model_dump(),
179 "commands_sent": len(responses),
180 "note": "Retune ausgeloest -- kurze RX-Unterbrechung, applied_seq zieht "
181 "verzoegert nach.",
182 }
