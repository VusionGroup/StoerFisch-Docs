57std::string IqRecorder::start(const std::string& dir,
58 std::uint32_t sample_rate_hz,
59 std::uint64_t center_freq_hz) {
60 // Falls schon eine Aufnahme läuft: sauber beenden, bevor eine neue beginnt.
61 stop();
62
63 const std::string basename = makeBasename(center_freq_hz);
64 // dir kann mit oder ohne Trailing-Slash kommen -- normalisieren.
65 std::string d = dir;
66 if (!d.empty() && d.back() != '/') d += '/';
67 base_path_ = d + basename;
68
69 const std::string cs8_path = base_path_ + ".cs8";
70 file_ = std::fopen(cs8_path.c_str(), "wb");
71 if (file_ == nullptr) {
72 // Verzeichnis nicht schreibbar / existiert nicht -- Aufrufer bekommt
73 // leeren String und kann das dem Nutzer melden.
74 base_path_.clear();
75 return {};
76 }
77
78 sample_rate_hz_ = sample_rate_hz;
79 center_freq_hz_ = center_freq_hz;
80 start_unix_ns_ = unixNs();
81 bytes_written_.store(0, std::memory_order_relaxed);
82 bytes_dropped_.store(0, std::memory_order_relaxed);
83 accepted_bytes_.store(0, std::memory_order_relaxed);
84 drop_event_count_.store(0, std::memory_order_relaxed);
85 drop_events_overflow_.store(0, std::memory_order_relaxed);
86
87 // Ring leeren (Indizes zurücksetzen), BEVOR der Producer feed() sieht.
88 read_.store(0, std::memory_order_relaxed);
89 write_.store(0, std::memory_order_relaxed);
90
91 // Eine erste .meta schreiben, damit die Aufnahme sofort selbstbeschreibend
92 // ist (sample_count wird beim Stop aktualisiert).
93 writeMeta();
94
95 stop_requested_.store(false, std::memory_order_relaxed);
96 // active_ mit Release setzen: Der Producer (feed) darf erst dann Bytes
97 // ablegen, wenn Datei und Indizes stehen.
98 active_.store(true, std::memory_order_release);
99 writer_ = std::thread(&IqRecorder::writerLoop, this);
100
101 return basename;
102}

104void IqRecorder::stop() {
105 if (!active_.load(std::memory_order_acquire)) {
106 return; // idempotent
107 }
108 // Producer stoppen: ab jetzt legt feed() nichts mehr ab.
109 active_.store(false, std::memory_order_release);
110 // Consumer signalisieren, dass er nach dem Leeren aussteigen soll.
111 stop_requested_.store(true, std::memory_order_release);
112 if (writer_.joinable()) {
113 writer_.join();
114 }
115
116 if (file_ != nullptr) {
117 std::fflush(file_);
118 std::fclose(file_);
119 file_ = nullptr;
120 }
121 // Finale .meta mit endgültigem sample_count/dropped.
122 writeMeta();
123 base_path_.clear();
124}

126void IqRecorder::feed(const std::int8_t* data, std::size_t bytes) noexcept {
127 if (!active_.load(std::memory_order_acquire) || bytes == 0) {
128 return;
129 }
130
131 const std::size_t w = write_.load(std::memory_order_relaxed);
132 const std::size_t r = read_.load(std::memory_order_acquire);
133 // Freier Platz im Ring: capacity - 1 nutzbar (ein Slot bleibt frei, um
134 // voll von leer zu unterscheiden).
135 const std::size_t used = (w - r) & mask_;
136 const std::size_t free = capacity_ - 1 - used;
137
138 if (bytes > free) {
139 // Ring würde überlaufen -> Schreib-Thread kommt nicht hinterher.
140 // Diesen Block komplett verwerfen (nicht teilweise -- ein halber Block
141 // würde die I/Q-Ausrichtung zerstören) und zählen.
142 bytes_dropped_.fetch_add(bytes, std::memory_order_relaxed);
143
144 // Position der Lücke festhalten: Sie sitzt nach allen bisher
145 // AKZEPTIERTEN Samples (accepted_bytes_/2), denn genau die landen in
146 // der Datei. So kann der Leser die Aufnahmezeit ab dieser Stelle um die
147 // Lückendauer korrigieren.
148 const std::uint64_t at_sample =
149 accepted_bytes_.load(std::memory_order_relaxed) / 2;
150 const std::size_t idx = drop_event_count_.load(std::memory_order_relaxed);
151 if (idx < kMaxDropEvents) {
152 drop_events_[idx].at_sample = at_sample;
153 drop_events_[idx].missing_samples = bytes / 2;
154 // Release: der Hauptthread (writeMeta) liest count mit Acquire und
155 // sieht dann garantiert den vollständig geschriebenen Eintrag.
156 drop_event_count_.store(idx + 1, std::memory_order_release);
157 } else {
158 // Kein Platz mehr für Details -- nur noch die Gesamtzahl der nicht
159 // mehr einzeln erfassten Lücken hochzählen.
160 drop_events_overflow_.fetch_add(1, std::memory_order_relaxed);
161 }
162 return;
163 }
164
165 // Block wird akzeptiert -> Positionszähler mitführen.
166 accepted_bytes_.fetch_add(bytes, std::memory_order_relaxed);
167
168 // Bytes in den Ring kopieren, ggf. mit Umbruch am Ende.
169 const std::size_t first = std::min(bytes, capacity_ - w);
170 std::memcpy(&ring_[w], data, first);
171 if (first < bytes) {
172 std::memcpy(&ring_[0], data + first, bytes - first);
173 }
174 // Release: der Consumer muss die kopierten Bytes sehen, bevor er den neuen
175 // write_-Wert liest.
176 write_.store((w + bytes) & mask_, std::memory_order_release);
177}

179void IqRecorder::writerLoop() {
180 while (true) {
181 const std::size_t r = read_.load(std::memory_order_relaxed);
182 const std::size_t w = write_.load(std::memory_order_acquire);
183 const std::size_t available = (w - r) & mask_;
184
185 if (available == 0) {
186 if (stop_requested_.load(std::memory_order_acquire)) {
187 break; // Ring leer und Stop angefordert -> fertig.
188 }
189 // Nichts zu tun -- kurz schlafen, statt die CPU zu verbrennen.
190 std::this_thread::sleep_for(std::chrono::milliseconds(2));
191 continue;
192 }
193
194 // So viel wie möglich am Stück schreiben: bis zum Ring-Ende (wegen des
195 // Umbruchs) bzw. bis zum Producer-Index. fwrite direkt aus dem Ring --
196 // kein Zwischenpuffer nötig, der Ring IST schon zusammenhängender
197 // Speicher.
198 const std::size_t contiguous = std::min(available, capacity_ - r);
199 const std::size_t written = std::fwrite(&ring_[r], 1, contiguous, file_);
200 bytes_written_.fetch_add(written, std::memory_order_relaxed);
201 // Release: der Producer darf den freigewordenen Platz erst sehen,
202 // nachdem wir aus dem Ring gelesen haben.
203 read_.store((r + contiguous) & mask_, std::memory_order_release);
204
205 if (written < contiguous) {
206 // Schreibfehler (Platte voll?) -- weiter versuchen wäre sinnlos.
207 // Der Producer läuft weiter, die Bytes gelten als "geschrieben"
208 // gezählt nur soweit written; der Rest geht verloren. Wir brechen
209 // NICHT ab, damit stop() den Thread noch sauber joinen kann, aber
210 // markieren über dropped, dass etwas fehlt.
211 bytes_dropped_.fetch_add(contiguous - written, std::memory_order_relaxed);
212 }
213 }
214 std::fflush(file_);
215}

217void IqRecorder::writeMeta() {
218 if (base_path_.empty()) {
219 return;
220 }
221 const std::string meta_path = base_path_ + ".cs8.meta";
222 std::FILE* mf = std::fopen(meta_path.c_str(), "wb");
223 if (mf == nullptr) {
224 return;
225 }
226 const std::uint64_t written = bytes_written_.load(std::memory_order_relaxed);
227 const std::uint64_t dropped = bytes_dropped_.load(std::memory_order_relaxed);
228 const std::uint64_t sample_count = written / 2;
229
230 // Acquire
231 const std::size_t n_gaps = drop_event_count_.load(std::memory_order_acquire);
232 const std::uint64_t gaps_overflow = drop_events_overflow_.load(std::memory_order_relaxed);
233
234 std::fprintf(mf,
235 "{\n"
236 " \"core:datatype\": \"ci8\",\n"
237 " \"core:sample_rate\": %u,\n"
238 " \"core:frequency\": %llu,\n"
239 " \"core:start_unix_ns\": %llu,\n"
240 " \"sample_count\": %llu,\n"
241 " \"bytes_dropped\": %llu,\n",
242 sample_rate_hz_,
243 static_cast<unsigned long long>(center_freq_hz_),
244 static_cast<unsigned long long>(start_unix_ns_),
245 static_cast<unsigned long long>(sample_count),
246 static_cast<unsigned long long>(dropped));
247
248 // Lücken (durch Ring-Überlauf verworfene Sample-Bereiche). Jede Lücke sitzt
249 // NACH at_sample geschriebenen Samples und ist missing_samples groß. Der
250 // Leser rekonstruiert die wahre Zeitachse: bis at_sample gilt die
251 // Startzeit + n/rate; ab dort kommt (missing_samples / rate) Sekunden
252 // hinzu, bevor es weitergeht.
253 std::fprintf(mf, " \"gaps_truncated\": %s,\n",
254 gaps_overflow > 0 ? "true" : "false");
255 std::fprintf(mf, " \"gaps\": [");
256 for (std::size_t i = 0; i < n_gaps; ++i) {
257 std::fprintf(mf, "\n {\"at_sample\": %llu, \"missing_samples\": %llu}%s",
258 static_cast<unsigned long long>(drop_events_[i].at_sample),
259 static_cast<unsigned long long>(drop_events_[i].missing_samples),
260 (i + 1 < n_gaps) ? "," : "");
261 }
262 std::fprintf(mf, "%s]\n}\n", n_gaps > 0 ? "\n " : "");
263 std::fclose(mf);
264}
