3Der Core nimmt genau EINEN Client an und beantwortet Requests streng
4sequentiell (40 Byte rein, 32 Byte raus). Wir serialisieren Zugriffe deshalb
5ueber ein asyncio.Lock -- so kann eine langsame Antwort keine zweite Anfrage

78 async def connect(self) -> None:
79 """Oeffnet die Unix-Socket-Verbindung zum Core."""
80 self._reader, self._writer = await asyncio.open_unix_connection(self._path)
81 self._alive = True
82
83 async def request(self, frame: bytes) -> cmd.CommandResponse:
84 """Sendet einen Frame und liest die Antwort -- unter Lock, damit sich
85 Anfragen nicht ueberholen. EOF beim Lesen => Core tot => CommandTransportError."""
86 if not self._alive or self._reader is None or self._writer is None:
87 raise CommandTransportError("Socket nicht verbunden")
88
89 async with self._lock:
90 try:
91 self._writer.write(frame)
92 await self._writer.drain()
93 raw = await self._reader.readexactly(cmd.RESPONSE_SIZE)
94 except (asyncio.IncompleteReadError, ConnectionError, OSError) as exc:
95 self._alive = False
96 raise CommandTransportError("Core hat die Verbindung geschlossen (EOF)") from exc
97
98 response = cmd.parse_response(raw)
99 self._generation = response.generation
100 return response
101
102 async def close(self) -> None:
103 self._alive = False
104 if self._writer is not None:
105 self._writer.close()
106 with contextlib.suppress(ConnectionError, OSError):
107 await self._writer.wait_closed()
108 self._writer = None
109 self._reader = None

136 async def request(self, frame: bytes) -> cmd.CommandResponse:
137 """Validiert wie der Core und reicht angenommene Kommandos an die
138 Fake-Telemetrie weiter (damit applied_command_seq nachzieht)."""
139 parsed = cmd.parse_request(frame)
140 seq = int(parsed["command_seq"])
141 kind = int(parsed["kind"])
142
143 status = self._validate(parsed)
144 if status == cmd.CommandStatus.OK:
145 self._telemetry.apply_command(
146 seq,
147 kind,
148 tx_gain_db=float(parsed["tx_gain_db"]),
149 target_channel=int(parsed["target_channel"]),
150 pattern_id=int(parsed["pattern_id"]),
151 overflow_policy=int(parsed["overflow_policy"]),
152 sample_rate_hz=int(parsed["sample_rate_hz"]),
153 center_freq_hz=int(parsed["center_freq_hz"]),
154 )
155 return cmd.CommandResponse(command_seq=seq, status=status, generation=self._generation)
156
157 def _validate(self, parsed: dict[str, Any]) -> cmd.CommandStatus:
158 """Zustandslose Validierung wie ``validateCommand()`` im Core: unbekannter
159 Kind, Kanal-/Gain-/Policy-/Retune-/Jamming-/Replay-Grenzen. Haelt Fake und
160 echten Core deckungsgleich, damit Tests gegen den Fake aussagekraeftig sind."""
161 kind = int(parsed["kind"])
162 try:
163 kind_enum = cmd.CommandKind(kind)
164 except ValueError:
165 return cmd.CommandStatus.UNKNOWN_KIND
166
167 if (kind_enum == cmd.CommandKind.SET_TARGET
168 and int(parsed["target_channel"]) > cmd.MAX_BLE_CHANNEL):
169 return cmd.CommandStatus.INVALID_CHANNEL
170 if kind_enum == cmd.CommandKind.SET_TX_GAIN and not (
171 cmd.MIN_TX_GAIN_DB <= float(parsed["tx_gain_db"]) <= cmd.MAX_TX_GAIN_DB
172 ):
173 return cmd.CommandStatus.INVALID_GAIN
174 if (kind_enum == cmd.CommandKind.SET_OVERFLOW_POLICY
175 and int(parsed["overflow_policy"]) not in (0, 1)):
176 return cmd.CommandStatus.INVALID_POLICY
177 # Retune-Kommandos: Grenzen wie im Core.
178 if (kind_enum == cmd.CommandKind.SET_SAMPLE_RATE and not (
179 MIN_SAMPLE_RATE_HZ <= int(parsed["sample_rate_hz"]) <= MAX_SAMPLE_RATE_HZ)):
180 return cmd.CommandStatus.INVALID_SAMPLE_RATE
181 if (kind_enum == cmd.CommandKind.SET_CENTER_FREQ and not (
182 MIN_CENTER_FREQ_HZ <= int(parsed["center_freq_hz"]) <= MAX_CENTER_FREQ_HZ)):
183 return cmd.CommandStatus.INVALID_FREQUENCY
184 # Jamming: Kanal 0..39, Muster 0 (Traeger) oder 1 (Rauschen).
185 if kind_enum == cmd.CommandKind.START_JAMMING:
186 if int(parsed["target_channel"]) > cmd.MAX_BLE_CHANNEL:
187 return cmd.CommandStatus.INVALID_CHANNEL
188 if int(parsed["pattern_id"]) not in (0, 1):
189 return cmd.CommandStatus.INVALID_PATTERN
190 # Replay: Kanal 0..39 ODER der Sentinel REPLAY_KEEP_ORIGINAL.
191 if kind_enum == cmd.CommandKind.START_REPLAY:
192 ch = int(parsed["target_channel"])
193 if ch != REPLAY_KEEP_ORIGINAL and ch > cmd.MAX_BLE_CHANNEL:
194 return cmd.CommandStatus.INVALID_CHANNEL
195 # RECORDING/STOP_TRANSMIT: parameterlos -> nichts zustandslos zu pruefen.
196 return cmd.CommandStatus.OK
