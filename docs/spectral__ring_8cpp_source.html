41SpectralRing::SpectralRing(const std::string& shm_name,
42 std::uint64_t capacity_slots,
43 std::uint32_t fft_size,
44 OverflowPolicy initial_policy)
45 : shm_name_(shm_name) {
46
47 if (capacity_slots == 0 || (capacity_slots & (capacity_slots - 1)) != 0) {
48 throw std::runtime_error("capacity_slots muss eine Potenz von 2 sein");
49 }
50 if (fft_size == 0 || (fft_size % 2) != 0) {
51 // Ungerade fft_size wuerde den Stride auf ein Nicht-Vielfaches von 8
52 // fuehren; align8() faengt das zwar ab, aber eine ungerade Bin-Zahl
53 // deutet auf einen Konfigurationsfehler hin -- FFT-Groessen sind in
54 // der Praxis Zweierpotenzen.
55 throw std::runtime_error("fft_size muss > 0 und geradzahlig sein");
56 }
57
58 const std::size_t stride = align8(sizeof(SpectralSlotHeader) +
59 static_cast<std::size_t>(fft_size) * sizeof(float));
60 mapped_size_ = sizeof(SpectralRingHeader) + capacity_slots * stride;
61
62 // Leiche eines abgestuerzten Cores entfernen; ENOENT ist der Normalfall.
63 // Ein noch angehaengter Consumer behaelt sein Mapping (siehe
64 // TelemetryRing-Konstruktor, gleiche Begruendung).
65 ::shm_unlink(shm_name_.c_str());
66
67 fd_ = ::shm_open(shm_name_.c_str(), O_CREAT | O_EXCL | O_RDWR, 0660);
68 if (fd_ < 0) {
69 throw std::runtime_error(
70 "shm_open fehlgeschlagen für " + shm_name_ +
71 " (laeuft bereits ein zweiter Core auf demselben Segment?)");
72 }
73
74 if (::ftruncate(fd_, static_cast<off_t>(mapped_size_)) != 0) {
75 ::close(fd_);
76 ::shm_unlink(shm_name_.c_str());
77 throw std::runtime_error("ftruncate fehlgeschlagen für " + shm_name_);
78 }
79
80 mapped_base_ = ::mmap(nullptr, mapped_size_, PROT_READ | PROT_WRITE, MAP_SHARED, fd_, 0);
81 if (mapped_base_ == MAP_FAILED) {
82 mapped_base_ = nullptr;
83 ::close(fd_);
84 ::shm_unlink(shm_name_.c_str());
85 throw std::runtime_error("mmap fehlgeschlagen für " + shm_name_);
86 }
87
88 header_ = static_cast<SpectralRingHeader*>(mapped_base_);
89 slots_base_ = static_cast<std::uint8_t*>(mapped_base_) + sizeof(SpectralRingHeader);
90
91 header_->schema_version = kSpectralSchemaVersion;
92 header_->capacity_slots = capacity_slots;
93 header_->generation = nowWallClockNs();
94 header_->fft_size = fft_size;
95 header_->record_stride = static_cast<std::uint32_t>(stride);
96 header_->overflow_policy.store(static_cast<std::uint32_t>(initial_policy),
97 std::memory_order_relaxed);
98 header_->reserved = 0;
99 header_->write_index.store(0, std::memory_order_relaxed);
100 header_->read_index.store(0, std::memory_order_relaxed);
101 header_->dropped_newest_count.store(0, std::memory_order_relaxed);
102 header_->dropped_oldest_count.store(0, std::memory_order_relaxed);
103
104 // Alle Slot-Seqlocks auf 0 (= gerade = "konsistent, aber leer"). Frisch
105 // per ftruncate angelegter Speicher ist ohnehin nullgefuellt; explizit,
106 // damit die Invariante nicht von dieser Zusicherung abhaengt.
107 for (std::uint64_t i = 0; i < capacity_slots; ++i) {
108 slotAt(i)->seq.store(0, std::memory_order_relaxed);
109 }
110
111 // Publikation: ab hier ist das Segment fuer Consumer gueltig (Release
112 // stellt sicher, dass alle Felder oben vorher sichtbar sind).
113 header_->magic.store(kSpectralMagic, std::memory_order_release);
114}

136bool SpectralRing::push(const float* bins,
137 std::size_t bin_count,
138 std::uint64_t timestamp_ns,
139 double center_freq_hz) noexcept {
140 if (bins == nullptr || bin_count != header_->fft_size) {
141 return false; // Programmierfehler des Aufrufers, kein Ringzustand
142 }
143
144 const std::uint64_t capacity = header_->capacity_slots;
145 // write_index: wir sind der einzige Schreiber -> relaxed genuegt.
146 const std::uint64_t w = header_->write_index.load(std::memory_order_relaxed);
147 // read_index: schreibt der Consumer -> acquire, damit wir seinen
148 // Fortschritt sehen, bevor wir ueber den Fuellstand entscheiden.
149 const std::uint64_t r = header_->read_index.load(std::memory_order_acquire);
150
151 // Monotone Indizes: die Differenz IST der Fuellstand, ohne Modulo-Tricks
152 // und ohne einen geopferten Slot.
153 const std::uint64_t in_flight = w - r;
154
155 if (in_flight >= capacity) {
156 // Ring voll. Policy bei jedem push() frisch lesen, damit ein
157 // Umschalten zur Laufzeit sofort wirkt.
158 const auto policy = static_cast<OverflowPolicy>(
159 header_->overflow_policy.load(std::memory_order_relaxed));
160
161 if (policy == OverflowPolicy::DropNewest) {
162 header_->dropped_newest_count.fetch_add(1, std::memory_order_relaxed);
163 return false;
164 }
165 // DropOldest: Wir ueberschreiben gleich einen ungelesenen Slot. Der
166 // read_index wird BEWUSST NICHT vom Producer verschoben -- das haette
167 // zwei Schreiber auf einer Variablen und damit die SPSC-Annahme
168 // gebrochen. Stattdessen erkennt der Consumer die Ueberrundung selbst
169 // an (write - read) > capacity und springt vor. Der Producer zaehlt
170 // hier nur mit, damit die Verlustinformation auch dann vorliegt, wenn
171 // gar kein Consumer angehaengt ist.
172 header_->dropped_oldest_count.fetch_add(1, std::memory_order_relaxed);
173 }
174
175 SpectralSlotHeader* slot = slotAt(w % capacity);
176
177 // --- Seqlock-Schreibprotokoll (Lamport) ---
178 // Wir sind der einzige Schreiber, also kennen wir den aktuellen Wert und
179 // koennen ihn relaxed lesen.
180 const std::uint32_t s = slot->seq.load(std::memory_order_relaxed);
181
182 // 1) Auf UNGERADE setzen = "Schreiben laeuft".
183 slot->seq.store(s + 1, std::memory_order_relaxed);
184 // Der Fence ist hier zwingend und nicht durch einen Release-Store am
185 // seq ersetzbar: Ein Release-Store verhindert nur, dass FRUEHERE
186 // Operationen nach hinten wandern -- er verhindert NICHT, dass die
187 // nachfolgenden Datenschreibvorgaenge VOR das seq-Update gezogen werden.
188 // Genau das muss aber ausgeschlossen sein, sonst sieht der Consumer
189 // veraenderte Daten bei noch geradem seq und haelt sie fuer konsistent.
190 std::atomic_thread_fence(std::memory_order_release);
191
192 // 2) Nutzdaten schreiben.
193 slot->sequence = static_cast<std::uint32_t>(w);
194 slot->timestamp_ns = timestamp_ns;
195 slot->center_freq_hz = center_freq_hz;
196 std::memcpy(binsOf(slot), bins, bin_count * sizeof(float));
197
198 // 3) Auf GERADE (+2) setzen = "fertig, neue Version". Der Fence davor
199 // garantiert, dass alle Datenschreibvorgaenge sichtbar sind, bevor
200 // der Consumer den geraden Wert sehen kann.
201 std::atomic_thread_fence(std::memory_order_release);
202 slot->seq.store(s + 2, std::memory_order_relaxed);
203
204 // Erst jetzt den Frame veroeffentlichen.
205 header_->write_index.store(w + 1, std::memory_order_release);
206 return true;
207}

243SpectralRingReader::SpectralRingReader(const std::string& shm_name) {
244 fd_ = ::shm_open(shm_name.c_str(), O_RDWR, 0);
245 if (fd_ < 0) {
246 throw std::runtime_error("shm_open (Reader) fehlgeschlagen für " + shm_name);
247 }
248
249 struct stat st {};
250 if (::fstat(fd_, &st) != 0 ||
251 static_cast<std::size_t>(st.st_size) < sizeof(SpectralRingHeader)) {
252 ::close(fd_);
253 throw std::runtime_error("Segment " + shm_name + " ist kleiner als der Header "
254 "-- Producer noch nicht fertig?");
255 }
256 mapped_size_ = static_cast<std::size_t>(st.st_size);
257
258 mapped_base_ = ::mmap(nullptr, mapped_size_, PROT_READ | PROT_WRITE, MAP_SHARED, fd_, 0);
259 if (mapped_base_ == MAP_FAILED) {
260 mapped_base_ = nullptr;
261 ::close(fd_);
262 throw std::runtime_error("mmap (Reader) fehlgeschlagen für " + shm_name);
263 }
264
265 header_ = static_cast<SpectralRingHeader*>(mapped_base_);
266
267 // magic ist das Publikations-Flag des Producers (Release-Store) -- der
268 // Acquire-Load hier stellt sicher, dass wir danach alle anderen
269 // Header-Felder vollstaendig sehen.
270 if (header_->magic.load(std::memory_order_acquire) != kSpectralMagic) {
271 ::munmap(mapped_base_, mapped_size_);
272 ::close(fd_);
273 throw std::runtime_error("Segment " + shm_name +
274 ": magic nicht gesetzt (Producer initialisiert gerade?)");
275 }
276 if (header_->schema_version != kSpectralSchemaVersion) {
277 const std::uint32_t found = header_->schema_version;
278 ::munmap(mapped_base_, mapped_size_);
279 ::close(fd_);
280 throw std::runtime_error("Schema-Version mismatch für " + shm_name + ": erwartet " +
281 std::to_string(kSpectralSchemaVersion) + ", gefunden " +
282 std::to_string(found));
283 }
284
285 slots_base_ = static_cast<const std::uint8_t*>(mapped_base_) + sizeof(SpectralRingHeader);
286 // Ab dem aktuellen Stand lesen, nicht ab 0: Alles davor liegt evtl. schon
287 // ausserhalb des Rings, und ein frisch angehaengter Consumer will die
288 // Gegenwart sehen, nicht die Historie.
289 read_pos_ = header_->write_index.load(std::memory_order_acquire);
290 header_->read_index.store(read_pos_, std::memory_order_release);
291}

308bool SpectralRingReader::tryPop(SpectralFrame& out) noexcept {
309 const std::uint64_t capacity = header_->capacity_slots;
310 const std::uint64_t w = header_->write_index.load(std::memory_order_acquire);
311
312 if (read_pos_ >= w) {
313 return false; // leer
314 }
315
316 // Ueberrundung: Der Producer ist um mehr als capacity vorausgeeilt, die
317 // aeltesten Frames existieren physisch nicht mehr. Auf den aeltesten noch
318 // vorhandenen Frame springen und den Verlust zaehlen.
319 if (w - read_pos_ > capacity) {
320 lapped_count_ += (w - read_pos_) - capacity;
321 read_pos_ = w - capacity;
322 }
323
324 const SpectralSlotHeader* slot = slotAt(read_pos_ % capacity);
325
326 // --- Seqlock-Leseprotokoll (Lamport) ---
327 const std::uint32_t s1 = slot->seq.load(std::memory_order_relaxed);
328 if ((s1 & 1u) != 0u) {
329 // Ungerade = Producer schreibt genau diesen Slot gerade. Bei
330 // drop-oldest heisst das: unser Frame wird uns gerade unter den
331 // Fingern ueberschrieben -- er ist verloren, ueberspringen. Ein
332 // Retry wuerde bei schnellem Producer endlos drehen, ohne dass die
333 // gesuchten Daten je zurueckkaemen.
334 ++torn_count_;
335 ++read_pos_;
336 header_->read_index.store(read_pos_, std::memory_order_release);
337 return false;
338 }
339
340 // Acquire-Fence: alle folgenden Datenlesevorgaenge duerfen nicht VOR den
341 // seq-Load gezogen werden.
342 std::atomic_thread_fence(std::memory_order_acquire);
343
344 SpectralFrame candidate;
345 candidate.sequence = slot->sequence;
346 candidate.timestamp_ns = slot->timestamp_ns;
347 candidate.center_freq_hz = slot->center_freq_hz;
348 candidate.bins.resize(header_->fft_size);
349 std::memcpy(candidate.bins.data(),
350 reinterpret_cast<const std::uint8_t*>(slot) + sizeof(SpectralSlotHeader),
351 static_cast<std::size_t>(header_->fft_size) * sizeof(float));
352
353 // Zweiter Fence + erneuter seq-Load: Hat sich seq veraendert, hat der
354 // Producer waehrend unserer Kopie geschrieben -> die Kopie ist ein
355 // Mischmasch aus altem und neuem Frame und muss verworfen werden.
356 std::atomic_thread_fence(std::memory_order_acquire);
357 const std::uint32_t s2 = slot->seq.load(std::memory_order_relaxed);
358
359 ++read_pos_;
360 header_->read_index.store(read_pos_, std::memory_order_release);
361
362 if (s1 != s2) {
363 ++torn_count_;
364 return false;
365 }
366
367 out = std::move(candidate);
368 return true;
369}
