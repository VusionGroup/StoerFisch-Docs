36@dataclass(frozen=True)
37class Config:
38 fake: bool = False
39 """Synthetische Quellen statt Shmem/Socket, es braucht keinen Core."""

45 shm_spectral: str = "vusion_spectral"

47 command_socket: str = "/tmp/vusion_cmd.sock"

49 # Spektrum: Push-Rate ans Canvas und Quantisierungsfenster.
50 spectrum_fps: float = 30.0
51 db_floor: float = -100.0
52 db_ceil: float = -20.0
53 # Wie viele Slots pro Poll maximal geleert werden. read_frame() liefert None
54 # sowohl fuer "leer" als auch fuer torn/lapped, deshalb eine harte Obergrenze
55 # statt einer Schleife bis "leer".
56 spectral_max_drain: int = 512
57 # Notch gegen den LO-Leakage-Spike des HackRF im Center-Bin, der sonst als
58 # konstanter Mittelstrich im Wasserfall steht. Halbe Fensterbreite in Bins:
59 # 1 interpoliert Center +/-1, 0 schaltet den Notch ab (dann bleibt der Spike
60 # als "Empfaenger lebt"-Referenz sichtbar). Kosmetik vor der Quantisierung.
61 dc_notch_half_width: int = 1
62
63 # Telemetrie-Poll, schnell genug damit dropped_count nicht steigt.
64 telemetry_poll_hz: float = 200.0
65 # Push-Rate an die Browser, absichtlich vom Poll entkoppelt: der echte Core
66 # emittiert deutlich schneller als die im UI sinnvollen 2-20 Hz. Ohne die
67 # Drossel flutet der Fan-out den Browser-Hauptthread mit JSON und Rerenders,
68 # requestAnimationFrame kommt nicht mehr dran und alle Canvas ruckeln.
69 # Snapshot-Kinds werden zwischen zwei Pushes auf den neuesten Wert
70 # zusammengefasst, diskrete Events (Detection/Safety) laufen ungedrosselt.
71 telemetry_push_hz: float = 20.0
72
73 # Dead-Man-Switch: bleibt der Heartbeat so lange aus, wird TX entschaerft.
74 deadman_timeout_s: float = 2.0
75
76 # Schreibrecht-Lease: bleibt der Heartbeat des Writers so lange aus, geht das
77 # Recht an den naechsten Wartenden. Grosszuegiger als der Dead-Man, weil hier
78 # nur "Verbindung tot" zaehlt und nicht die TX-Safety.
79 lease_timeout_s: float = 5.0
80
81 # Anwesenheits-Rueckfrage: so lange darf der Writer untaetig sein (kein
82 # Kommando, kein "noch da"-Klick), bevor bei weiteren Zuschauern nachgefragt
83 # wird -- und so lange hat er dann Zeit zu antworten.
84 lease_presence_interval_s: float = 180.0
85 lease_challenge_grace_s: float = 30.0
86
87 # Startwerte der SDR-Sicht. Achtung, das ist eine Annahme: der Core startet
88 # mit seinen CLI-Werten, und Rod erfaehrt beim Verbinden weder die
89 # sample_rate (steht nicht im Spektral-Ring) noch, womit der Core gestartet
90 # wurde. Beim Start senden wir bewusst kein SET_SAMPLE_RATE/SET_CENTER_FREQ,
91 # das waere ein HW-Retune samt RX-Luecke bei jedem Rod-Neustart. Also
92 # entweder diese Defaults passend zur Core-CLI setzen (ROD_SAMPLE_RATE_HZ /
93 # ROD_CENTER_FREQ_HZ) oder im SDR-Panel einmal "Setzen" druecken, danach ist
94 # die Sicht sicher synchron. Die Frequenzachse einzelner Frames kommt
95 # ohnehin aus frame.center_freq_hz und nicht von hier.
96 default_sample_rate_hz: float = 20_000_000.0
97 default_center_freq_hz: float = 2_440_000_000.0
