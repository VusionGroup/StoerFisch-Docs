14def _env_bool(name: str, default: bool) -> bool:
15 """Liest eine ROD_*-Env als bool (1/true/yes/on = True)."""
16 raw = os.environ.get(name)
17 if raw is None:
18 return default
19 return raw.strip().lower() in {"1", "true", "yes", "on"}

40@dataclass(frozen=True)
41class Config:
42 """Zusammengefasste Laufzeitkonfiguration."""
43
44 fake: bool = False
45 """True -> synthetische Quellen statt Shmem/Socket (kein Core noetig)."""
46
47 host: str = "0.0.0.0"
48 port: int = 8000
49
50 # Ressourcennamen (Defaults passend zur Core-CLI).
51 shm_spectral: str = "vusion_spectral"
52 shm_telemetry: str = "vusion_telemetry"
53 command_socket: str = "/tmp/vusion_cmd.sock"
54
55 # Spektrum: Push-Rate ans Canvas und Quantisierungsfenster.
56 spectrum_fps: float = 30.0
57 db_floor: float = -100.0
58 db_ceil: float = -20.0
59 # Obergrenze, wie viele Slots pro Poll gedrained werden (Live -> neuester
60 # Frame). read_frame() gibt None fuer leer UND torn/lapped zurueck, deshalb
61 # begrenzen wir die Drain-Schleife hart, statt bis "leer" zu laufen.
62 spectral_max_drain: int = 512
63 # DC-/Center-Bin-Notch gegen den HackRF-LO-Leakage-Spike (konstanter
64 # Mittelstrich). Halbe Fensterbreite in Bins: 1 -> Center +/-1 (3 Bins)
65 # werden interpoliert, 0 -> Notch aus (Spike bleibt als "Empfaenger lebt"-
66 # Referenz sichtbar). Rein kosmetisch, greift erst vor der Quantisierung.
67 dc_notch_half_width: int = 1
68
69 # Telemetrie-Poll-Intervall (schnell genug, damit dropped_count nicht steigt).
70 telemetry_poll_hz: float = 200.0
71 # Push-Rate an die Browser. Bewusst ENTKOPPELT vom Poll: der echte Core
72 # emittiert Records schneller als die im UI sinnvollen 2-20 Hz. Ohne diese
73 # Drossel flutet der Fan-out den Browser-Hauptthread mit JSON+Rerenders und
74 # hungert requestAnimationFrame aus (alle Canvas ruckeln). Snapshot-Kinds
75 # werden zwischen zwei Pushes zum jeweils neuesten Wert zusammengefasst;
76 # diskrete Events (Detection/Safety) laufen ungedrosselt durch.
77 telemetry_push_hz: float = 20.0
78
79 # Dead-Man-Switch: ohne Heartbeat innerhalb dieser Zeit wird TX entschaerft.
80 deadman_timeout_s: float = 2.0
81
82 # Startwert der SDR-Konfiguration. ACHTUNG, das ist eine ANNAHME: Der Core
83 # startet mit seinen CLI-Werten, und Rod erfaehrt beim Verbinden weder
84 # sample_rate (steht nicht im Spektral-Ring) noch, womit der Core gestartet
85 # wurde. Wir senden hier bewusst KEIN SET_SAMPLE_RATE/SET_CENTER_FREQ beim
86 # Start -- das waere ein HW-Retune samt RX-Luecke bei jedem Rod-Neustart.
87 # Stattdessen: diese Defaults passend zur Core-CLI setzen (ROD_SAMPLE_RATE_HZ
88 # / ROD_CENTER_FREQ_HZ), oder im SDR-Panel einmal "Setzen" druecken -- danach
89 # ist die Sicht garantiert synchron. Die Frequenzachse einzelner Frames
90 # kommt ohnehin aus frame.center_freq_hz, nicht von hier.
91 default_sample_rate_hz: float = 20_000_000.0
92 default_center_freq_hz: float = 2_440_000_000.0
