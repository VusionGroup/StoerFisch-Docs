81Config parseArgs(int argc, char** argv) {
82 Config c;
83 for (int i = 1; i < argc; ++i) {
84 const std::string a = argv[i];
85 auto next = [&]() -> std::string {
86 if (i + 1 >= argc) { std::fprintf(stderr, "Fehlender Wert für %s\n", a.c_str()); std::exit(2); }
87 return argv[++i];
88 };
89 if (a == "--freq") c.freq_hz = std::stoull(next());
90 else if (a == "--sample-rate") c.sample_rate = std::stod(next());
91 else if (a == "--fft-size") c.fft_size = static_cast<std::uint32_t>(std::stoul(next()));
92 else if (a == "--lna-gain") c.lna_gain = static_cast<std::uint32_t>(std::stoul(next()));
93 else if (a == "--vga-gain") c.vga_gain = static_cast<std::uint32_t>(std::stoul(next()));
94 else if (a == "--spectral-shm") c.spectral_shm = next();
95 else if (a == "--telemetry-shm") c.telemetry_shm = next();
96 else if (a == "--socket") c.socket_path = next();
97 else if (a == "--enter-dbm") c.enter_dbm = std::stof(next());
98 else if (a == "--exit-dbm") c.exit_dbm = std::stof(next());
99 else if (a == "--recording-dir") c.recording_dir = next();
100 else if (a == "--replay-file") c.replay_file = next();
101 else { std::fprintf(stderr, "Unbekanntes Argument: %s\n", a.c_str()); std::exit(2); }
102 }
103 return c;
104}

146 // --- Sende-Zustand (TX) ---
147 // Der HackRF ist halbduplex: Jamming/Replay stoppen den RX. Der
148 // Hauptthread wechselt den Modus (HW-Aufrufe), der Callback legt nur
149 // Wuensche ab. Da im TX-Modus kein RX-Callback laeuft, pollt der
150 // Hauptthread die Kommandos dann selbst.
151 enum class TxWish : std::uint32_t { None = 0, Jam = 1, Replay = 2, Stop = 3 };
152 std::atomic<std::uint32_t> tx_wish{static_cast<std::uint32_t>(TxWish::None)};
153 std::atomic<std::uint32_t> tx_channel{0}; // Zielkanal (BLE) fuer Jam/Replay
154 std::atomic<std::uint32_t> tx_pattern{0}; // 0=CW/kein Loop, 1=Noise/Loop
155 bool tx_active = false; // laeuft gerade ein Sendebetrieb?
156 // Merkt sich die zuletzt gestartete Aufnahme, damit StartReplay ohne
157 // --replay-file "die letzte Aufnahme" abspielen kann.
158 std::string last_recording_path;

165 std::printf(" spectral=%s telemetry=%s socket=%s\n",

171 // Gemeinsame Kommando-Verarbeitung. Wird aus dem RX-Callback UND (im
172 // TX-Modus) aus dem Hauptthread aufgerufen. Setzt nur Flags/Wuensche --
173 // die eigentliche Aktion (HW, Datei) fuehrt der Hauptthread aus.
174 auto handle_command = [&](const CommandRequest& cmd) {
175 switch (static_cast<CommandKind>(cmd.kind)) {
176 case CommandKind::StartRecording:
177 want_start_recording.store(true, std::memory_order_release);
178 break;
179 case CommandKind::StopRecording:
180 want_stop_recording.store(true, std::memory_order_release);
181 break;
182 case CommandKind::StartJamming:
183 tx_channel.store(cmd.target_channel, std::memory_order_relaxed);
184 tx_pattern.store(cmd.pattern_id, std::memory_order_relaxed);
185 tx_wish.store(static_cast<std::uint32_t>(TxWish::Jam), std::memory_order_release);
186 break;
187 case CommandKind::StartReplay:
188 tx_channel.store(cmd.target_channel, std::memory_order_relaxed);
189 tx_pattern.store(cmd.pattern_id, std::memory_order_relaxed);
190 tx_wish.store(static_cast<std::uint32_t>(TxWish::Replay), std::memory_order_release);
191 break;
192 case CommandKind::StopTransmit:
193 tx_wish.store(static_cast<std::uint32_t>(TxWish::Stop), std::memory_order_release);
194 break;
195 default:
196 pipeline.applyCommand(cmd);
197 break;
198 }
199 };

201 // RX-Callback: laeuft im libusb-Kontext. Erst Kommandos einspeisen,
202 // dann den IQ-Block verarbeiten, dann (falls aktiv) die ROHBYTES in den
203 // Recorder fuettern. Alles ohne libhackrf-Aufrufe und ohne Blockieren.
204 auto rx_callback = [&](const IqBlock& block) {
205 const std::uint64_t t1 = monotonicNs();
206 CommandRequest cmd{};
207 while (channel.tryPopCommand(cmd)) {
208 handle_command(cmd);
209 }
210 pipeline.processIqBlock(block, t1);

212 // auf die originalen int8-I/Q-Bytes; num_samples komplexe Samples
213 // = 2*num_samples Bytes.
214 if (block.raw != nullptr) {

216 }
217 };

245 // Zielkanal -> TX-Center-Frequenz. Die Quellen erzeugen ein

247 // HW-Frequenz. (0xFFFFFFFF bei Replay = Originalfrequenz

284 loop ? " [Loop]" : "", ch);

289 std::fprintf(stderr, "TX-Start fehlgeschlagen: %s\n", e.what());
290 // Versuchen, in den RX-Betrieb zurueckzukehren.
291 try { sdr.setAmpEnable(false); sdr.startRx(rx_callback); } catch (...) {}
292 tx_active = false;
293 }
294 } else if (wish == TxWish::Stop && tx_active) {
295 try {

297 sdr.setAmpEnable(false);

309 // und das Ende eines nicht geloopten Replays erkennen. ---

345 }
346 }
347 if (want_stop_recording.exchange(false, std::memory_order_acq_rel)) {
348 if (recorder.isRecording()) {
349 const RecorderStats st = recorder.stats();
350 recorder.stop();

352 static_cast<unsigned long long>(st.bytes_written),
353 static_cast<unsigned long long>(st.bytes_dropped));
354 std::fflush(stdout);
355 }
356 }
357
358 SignalPipeline::PendingRetune rt;
359 if (pipeline.takePendingRetune(rt)) {
360 // Neue Zielwerte bestimmen (nur die tatsaechlich gesetzten
361 // Felder aendern sich).
362 if (rt.has_sample_rate) current_rate = static_cast<double>(rt.sample_rate_hz);
363 if (rt.has_center_freq) current_freq = static_cast<double>(rt.center_freq_hz);
364
365 std::printf("Retune: freq=%.0f Hz rate=%.0f Hz (seq=%u)\n",
366 current_freq, current_rate, rt.command_seq);
367 std::fflush(stdout);
368
369 // HW-Umkonfiguration OHNE Callback-Kontext: RX stoppen,
370 // umstellen, neu starten. Genau hier duerfen hackrf_set_*
371 // aufgerufen werden.
372 try {
373 sdr.stopRx();
374 if (rt.has_sample_rate) sdr.setSampleRate(current_rate);
375 if (rt.has_center_freq) sdr.setFrequency(static_cast<std::uint64_t>(current_freq));
376 // Analyzer/Telemetrie auf die neuen Werte ziehen und
377 // Ack-Ebene 3 melden -- waehrend RX noch steht.
378 pipeline.reconfigureAfterRetune(current_freq, current_rate, rt.command_seq);
379 // Recorder-Meta-Parameter mitziehen: eine kuenftige Aufnahme
380 // soll die neue Frequenz/Rate festhalten.
381 rec_rate_hz.store(static_cast<std::uint32_t>(current_rate), std::memory_order_release);
382 rec_freq_hz.store(static_cast<std::uint64_t>(current_freq), std::memory_order_release);
383 sdr.startRx(rx_callback);
384 } catch (const std::exception& e) {
385 // Retune fehlgeschlagen -- der Core bleibt am Leben, aber
386 // ohne laufendes RX koennen wir nicht weitermachen.
387 std::fprintf(stderr, "Retune fehlgeschlagen: %s\n", e.what());
388 break;
389 }
390 }
391 std::this_thread::sleep_for(std::chrono::milliseconds(20));
392 }
393
394 std::printf("\nShutdown angefordert, beende...\n");
395 recorder.stop(); // laufende Aufnahme sauber abschliessen
396 if (tx_active) {
397 sdr.stopTx();
398 sdr.setAmpEnable(false);
399 } else {
400 sdr.stopRx();
401 }
402 channel.stop();
403 spectral.destroy();
404 telemetry.destroy();
405 std::printf("Sauber beendet.\n");
406 return 0;
407
408 } catch (const std::exception& e) {
409 std::fprintf(stderr, "FATAL: %s\n", e.what());
410 return 1;
