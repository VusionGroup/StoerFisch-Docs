49 void SignalPipeline::applyCommand(const CommandRequest &cmd) {
50 switch (static_cast<CommandKind>(cmd.kind)) {
51 case CommandKind::Start:
52 controller_.start();
53 break;
54 case CommandKind::Stop:
55 controller_.stop();
56 break;
57 case CommandKind::SetTarget:
58 // Manuelles Ziel/Muster -- ueberschreibt hier nur die
59 // Muster-/Kanalfelder; die eigentliche Zielwahl macht der Regler.
60 actuator_.pattern_id = cmd.pattern_id;
61 break;
62 case CommandKind::SetTxGain:
63 actuator_.tx_gain_db = cmd.tx_gain_db;
64 break;
65 case CommandKind::SetOverflowPolicy:
66 spectral_.setOverflowPolicy(static_cast<OverflowPolicy>(cmd.overflow_policy));
67 break;
68 case CommandKind::SetSampleRate:
69 case CommandKind::SetCenterFreq: {
70 // NICHT hier die Hardware anfassen -- wir sind im RX-Callback, und
71 // hackrf_set_* darf da nicht aufgerufen werden. Nur den Wunsch
72 // ablegen; der Hauptthread führt den Retune aus.
73 //
74 // Ein evtl. schon vorliegender, noch nicht abgeholter Wunsch wird
75 // ergänzt (nicht überschrieben): So können SetSampleRate und
76 // SetCenterFreq kurz nacheinander kommen und werden in EINEM Retune
77 // zusammengefasst.
78 if (static_cast<CommandKind>(cmd.kind) == CommandKind::SetSampleRate) {
79 pending_retune_.has_sample_rate = true;
80 pending_retune_.sample_rate_hz = cmd.sample_rate_hz;
81 } else {
82 pending_retune_.has_center_freq = true;
83 pending_retune_.center_freq_hz = cmd.center_freq_hz;
84 }
85 pending_retune_.command_seq = cmd.command_seq;
86 // Release: stellt sicher, dass der Hauptthread beim Acquire-Lesen
87 // des Flags die oben geschriebenen Felder vollständig sieht.
88 has_pending_retune_.store(true, std::memory_order_release);
89 // WICHTIG: applied_command_seq NICHT hier setzen. Der Retune wirkt
90 // erst, wenn der Hauptthread ihn ausgeführt hat -- erst dann meldet
91 // reconfigureAfterRetune() das seq als angewendet (Ack-Ebene 3).
92 return;
93 }
94 case CommandKind::Ping:
95 case CommandKind::Shutdown:
96 break;
97 }
98 actuator_.applied_command_seq = cmd.command_seq;
99 }

101 void SignalPipeline::processIqBlock(const IqBlock &block, std::uint64_t t1_adc_ns) {
102 // --- 1) FFT ---
103 analyzer_.process(block, magnitudes_);
104 const std::uint64_t t2_fft_done = monotonicNs(); // FFT tatsaechlich fertig
105
106 // FFT-Magnituden in den Spektral-Ring (#1).
107 spectral_.push(magnitudes_.data(), magnitudes_.size(), t2_fft_done, center_freq_);
108
109 // --- 2) Kanalbelegung + Events ---
110 analyzer_.channelEnergies(channel_energies_);
111 const std::uint64_t mask = tracker_.update(channel_energies_, events_);
112
113 // Detektions-Events (#3) einzeln loggen.
114 for (const auto &ev: events_) {
115 DetectionEventPayload dp{};
116 dp.channel = ev.channel;
117 dp.event_kind = static_cast<std::uint32_t>(ev.kind);
118 dp.rssi_dbm = ev.energy_dbm;
119 pushTelemetry(RecordType::DetectionEvent, t1_adc_ns, dp);
120 }
121
122 // Kanal-Belegungs-Snapshot (#2).
123 ChannelOccupancyPayload cop{};
124 cop.channel_occupied_mask = mask;
125 for (std::uint32_t k = 0; k < kBleChannelCount; ++k) {
126 // Energie auf int8 (1-dB-Raster) begrenzen, wie im Payload definiert.
127 float e = channel_energies_[k];
128 if (e > 127.0f) e = 127.0f;
129 if (e < -128.0f) e = -128.0f;
130 cop.energy_dbm[k] = static_cast<std::int8_t>(e);
131 }
132 pushTelemetry(RecordType::ChannelOccupancy, t1_adc_ns, cop);
133
134 // --- 3) Regler-Entscheidung ---
135 const ControllerOutput out = controller_.decide(mask, channel_energies_);
136 const std::uint64_t t3_decision = monotonicNs();
137
138 // Aktorzustand aktualisieren.
139 actuator_.enabled = out.jammer_enabled;
140 if (out.jammer_enabled) {
141 actuator_.active_channel = out.target_channel;
142 }
143
144 // In der echten Version wuerde hier das TX-Stellsignal ausgegeben.
145 const std::uint64_t t4_dac_out = monotonicNs();
146
147 // --- 4) Telemetrie: Controller-Zustand (#5) ---
148 ControllerStatePayload csp{};
149 csp.target_channel = out.target_channel;
150 csp.observed_channel = out.target_channel; // Ist == Soll in dieser Basis-Strategie
151 csp.state = static_cast<std::uint32_t>(out.state);
152 csp.error_metric = out.error_metric;
153 csp.decision = static_cast<std::uint32_t>(out.decision);
154 pushTelemetry(RecordType::ControllerState, t3_decision, csp);
155
156 // --- 5) Telemetrie: Aktorzustand (#6), inkl. applied_command_seq ---
157 ActuatorStatePayload asp{};
158 asp.active_channel = actuator_.active_channel;
159 asp.tx_gain_db = actuator_.tx_gain_db;
160 asp.pattern_id = actuator_.pattern_id;
161 asp.applied_command_seq = actuator_.applied_command_seq;
162 asp.enabled = actuator_.enabled ? 1 : 0;
163 pushTelemetry(RecordType::ActuatorState, t4_dac_out, asp);
164
165 // --- 6) Telemetrie: Latenzkette (#4) ---
166 LatencyTimestampsPayload lat{};
167 lat.t1_adc_ns = t1_adc_ns;
168 lat.t2_fft_done_ns = t2_fft_done;
169 lat.t3_decision_ns = t3_decision;
170 lat.t4_dac_out_ns = t4_dac_out;
171 pushTelemetry(RecordType::LatencyTimestamps, t4_dac_out, lat);
172
173 // --- 7) Telemetrie: Buffer-Health (#8), nur alle N Zyklen ---
174 // Aus DIESEM Thread, nicht aus einem Housekeeping-Thread: Der
175 // TelemetryRing ist SPSC, ein zweiter Producer wäre ein Data Race auf
176 // write_index. (Genau dieser Fehler steckte zwischenzeitlich in
177 // main_loop.cpp und wurde behoben.)
178 if (++cycle_counter_ >= kBufferHealthInterval) {
179 cycle_counter_ = 0;
180 BufferHealthPayload bh{};
181 // usb_overruns/underruns: die libhackrf-Fehlerzähler sind noch nicht
182 // angebunden -- bewusst 0 statt einer erfundenen Zahl.
183 bh.usb_overruns = 0;
184 bh.usb_underruns = 0;
185 // Verluste im Spektralpfad, beide Policies zusammengefasst. Die
186 // Aufschlüsselung nach Policy steht im SpectralRing-Header, den Rod
187 // ohnehin direkt liest.
188 bh.dropped_samples = static_cast<std::uint32_t>(
189 spectral_.droppedNewestCount() + spectral_.droppedOldestCount());
190 bh.actual_sample_rate_hz = reported_sample_rate_hz_;
191 bh.pll_locked = 1; // ohne HW-Abfrage: Annahme, siehe Doku
192 pushTelemetry(RecordType::BufferHealth, t4_dac_out, bh);
193 }
194 }

196 bool SignalPipeline::takePendingRetune(PendingRetune &out) noexcept {
197 // Acquire: paart mit dem Release-Store in applyCommand -- sehen wir true,
198 // sehen wir auch die vollständig geschriebenen pending_retune_-Felder.
199 if (!has_pending_retune_.load(std::memory_order_acquire)) {
200 return false;
201 }
202 out = pending_retune_;
203 // Felder zurücksetzen, damit der nächste Retune frisch beginnt.
204 pending_retune_ = PendingRetune{};
205 has_pending_retune_.store(false, std::memory_order_release);
206 return true;
207 }

209 void SignalPipeline::reconfigureAfterRetune(double center_freq_hz,
210 double sample_rate_hz,
211 std::uint32_t command_seq) noexcept {
212 // Läuft im Hauptthread, während RX gestoppt ist -- process() läuft also
213 // nicht, der Analyzer-Zugriff ist unkritisch.
214 analyzer_.reconfigure(center_freq_hz, sample_rate_hz);
215 center_freq_ = center_freq_hz;
216 reported_sample_rate_hz_ = static_cast<float>(sample_rate_hz);
217 // Jetzt ist der Retune wirklich aktiv -> Ack-Ebene 3.
218 actuator_.applied_command_seq = command_seq;
