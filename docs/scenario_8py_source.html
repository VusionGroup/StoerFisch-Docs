36 @classmethod
37 def sweep_power(
38 cls, start_db: float, stop_db: float, step_db: float, dwell_s: float,
39 ) -> Scenario:
40 """Baut einen Leistungs-Sweep: TX-Gain von ``start_db`` bis ``stop_db`` in
41 ``step_db``-Stufen, je Stufe ``dwell_s`` Sekunden. Richtung ergibt sich aus
42 start/stop (auch abwaerts moeglich)."""
43 steps: list[Step] = []
44 direction = 1 if stop_db >= start_db else -1
45 step_mag = abs(step_db) or 1.0
46 gain = start_db
47 while (direction == 1 and gain <= stop_db) or (direction == -1 and gain >= stop_db):
48 steps.append(Step(duration_s=dwell_s, tx_gain_db=round(gain, 3),
49 note=f"sweep {gain:g} dB"))
50 gain += direction * step_mag
51 return cls(name=f"power_sweep_{start_db:g}_{stop_db:g}", steps=steps)
