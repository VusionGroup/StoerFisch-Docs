45def test_notch_dc_replaces_center_spike():
46 # Flacher -90-dB-Boden mit einem kuenstlichen Spike im Center-Bin (128).
47 bins = np.full(256, -90.0, dtype=np.float32)
48 bins[128] = -10.0
49 out = notch_dc(bins, half_width=1)
50 # Center +/-1 auf Nachbarniveau interpoliert -> Spike weg.
51 assert out[128] == pytest.approx(-90.0)
52 assert out[127] == pytest.approx(-90.0)
53 assert out[129] == pytest.approx(-90.0)
54 # Ausserhalb des Fensters unveraendert.
55 assert out[100] == pytest.approx(-90.0)
