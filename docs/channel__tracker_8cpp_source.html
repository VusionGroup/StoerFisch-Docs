24 std::uint64_t ChannelTracker::update(const std::vector<float>& channel_energies_dbm,
25 std::vector<ChannelEvent>& out_events) {
26 out_events.clear();
27 if (channel_energies_dbm.size() < kBleChannels) {
28 // Unerwartete Eingabelänge -- Zustand unverändert lassen statt raten.
29 return occupied_mask_;
30 }
31
32 for (std::uint32_t k = 0; k < kBleChannels; ++k) {
33 const float e = channel_energies_dbm[k];
34 const bool was_occupied = (occupied_mask_ >> k) & 1u;
35
36 if (!was_occupied && e >= enter_dbm_) {
37 // Freie Flanke: Kanal wird belegt.
38 occupied_mask_ |= (1ull << k);
39 out_events.push_back({k, ChannelEventKind::BecameOccupied, e});
40 } else if (was_occupied && e < exit_dbm_) {
41 // Fallende Flanke: Kanal wird frei.
42 occupied_mask_ &= ~(1ull << k);
43 out_events.push_back({k, ChannelEventKind::BecameFree, e});
44 }
45 // Dazwischen (exit_dbm_ <= e < enter_dbm_ bei belegt, oder e < enter
46 // bei frei): Zustand bleibt -- das ist die Hysterese.
47 }
48
49 return occupied_mask_;
50 }
