1"""Echter Spektral-Consumer: Wrapper um Cores's SpectralRingReader.
2
3Fuer Live-Monitoring unter drop-oldest wollen wir immer den **neuesten** Frame.
4``read_frame()`` rueckt aber nur einen Slot vor und gibt ``None`` sowohl bei
5"leer" als auch bei "torn/lapped" zurueck -- die drei Faelle sind am Rueckgabe-
6wert nicht unterscheidbar (das provided-Modul hat bewusst keine ``pending``-
7Property). Deshalb drainen wir in einer **hart gedeckelten** Schleife
8(``max_drain``) bis zu einer kurzen Leer-Serie und behalten den letzten
9gueltigen Frame. Die uebersprungenen zaehlen wir separat als
10``frames_dropped_in_drain`` -- Rods eigenes Verwerfen fuers Display, getrennt
11von den Ring-internen Verlustzaehlern.
12"""
13

22# bleibt stehen.

50 def read_latest(self, max_drain: int) -> SpectrumFrame | None:
51 """Drained bis zu ``max_drain`` Slots und gibt den NEUESTEN gueltigen Frame
52 zurueck (uebersprungene zaehlen als ``frames_dropped_in_drain``). Bricht
53 nach einer kurzen Leer-Serie ab. ``None``, wenn nichts Gueltiges kam.
54
55 Args:
56 max_drain: harte Obergrenze der Schleife -- verhindert, dass ein voll
57 nachlaufender Ring den Poll-Task blockiert.
58 """
59 latest: spec.SpectralFrame | None = None
60 none_streak = 0
61 for _ in range(max(1, max_drain)):
62 frame = self._reader.read_frame()
63 if frame is None:
64 none_streak += 1
65 if none_streak >= _EMPTY_STREAK:
66 break
67 continue
68 none_streak = 0
69 if latest is not None:
70 self._dropped_in_drain += 1
71 latest = frame
72 self._frames_read += 1
73
74 if latest is None:
75 return None
76 return SpectrumFrame(
77 sequence=latest.sequence,
78 timestamp_ns=latest.timestamp_ns,
79 center_freq_hz=latest.center_freq_hz,
80 bins=latest.bins,
81 )
