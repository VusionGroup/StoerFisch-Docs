1"""Synthetische Telemetrie fuer den Fake-Betrieb (ROD_FAKE=1).
2
3Erzeugt Records aller Typen und reagiert auf Kommandos: START/STOP schalten den
4Regelkreis, SET_TX_GAIN/SET_TARGET setzen Aktuator-Felder, und jedes angewandte
5Kommando spiegelt seine ``command_seq`` in ``applied_command_seq`` zurueck --
6damit der pending->applied-Weg (Ack-Ebene 3) auch ohne Core testbar ist.
7
8Der Fake bildet #7 SuccessMetric ab, obwohl der echte Core ihn derzeit nicht
9schreibt -- so laesst sich der UI-Pfad trotzdem ueben.
10
11SET_SAMPLE_RATE / SET_CENTER_FREQ werden bewusst MIT ihrer unangenehmen Seite
12nachgebildet: ein Retune stellt echte Hardware um, deshalb schweigt der Fake fuer
13``_RETUNE_GAP_S`` (RX-Unterbrechung) und meldet ``applied_command_seq`` erst
14danach. Waere das hier "sofort", wuerde Rod gegen ein zu freundliches Modell
15entwickelt und die Wartelogik am echten Core zum ersten Mal getestet.
16"""

33class FakeTelemetrySource:
34 """Synthetische ``TelemetrySource`` + Kommando-Senke fuer den Fake-Betrieb.
35
36 Haelt einen kleinen Simulationszustand (Regelkreis an/aus, Aktuator, Retune,
37 Sendebetrieb) und erzeugt daraus Records aller Typen. ``apply_command`` (vom
38 Fake-Transport aufgerufen) veraendert diesen Zustand -- so wird der komplette
39 Steuer->Wirkung-Weg (bis Ack-Ebene 3) ohne Core testbar.
40 """

48 Args:
49 sample_rate_hz: Start-Samplerate (fuer buffer_health / SDR-Sicht).
50 center_freq_hz: Start-Mittenfrequenz (fuer die SDR-Sicht).
51 """

54 self._last_cycle = 0.0

82 def apply_command(self, command_seq: int, kind: int, **fields: Any) -> None:
83 """Wendet ein validiertes Kommando auf den Sim-Zustand an."""
84 ck = tel_cmd()
85 if kind == int(ck.START):
86 self._running = True
87 elif kind == int(ck.STOP):
88 self._running = False
89 self._enabled = False
90 elif kind == int(ck.SET_TX_GAIN):
91 self._tx_gain_db = float(fields.get("tx_gain_db", self._tx_gain_db))
92 elif kind == int(ck.SET_TARGET):
93 self._pattern_id = int(fields.get("pattern_id", self._pattern_id))
94 elif kind in (int(ck.SET_SAMPLE_RATE), int(ck.SET_CENTER_FREQ)):
95 if kind == int(ck.SET_SAMPLE_RATE):
96 self._sample_rate_hz = float(fields.get("sample_rate_hz", self._sample_rate_hz))
97 else:
98 self._center_freq_hz = float(fields.get("center_freq_hz", self._center_freq_hz))
99 # Ack-Ebene 3 erst nach der Unterbrechung. Trifft ein zweiter Retune
100 # waehrend der Luecke ein, faellt er wie im Core mit dem ersten
101 # zusammen -- die Luecke verlaengert sich nicht, nur die Sequenz
102 # rueckt vor.
103 now = time.monotonic()
104 if self._retune_until is None:
105 self._retune_until = now + _RETUNE_GAP_S
106 self._retune_seq = command_seq
107 return
108 elif kind == int(ck.START_RECORDING):
109 self._recording = True
110 elif kind == int(ck.STOP_RECORDING):
111 self._recording = False
112 elif kind == int(ck.START_JAMMING):
113 self._transmitting = True
114 self._enabled = True
115 self._active_channel = int(fields.get("target_channel", self._active_channel))
116 self._pattern_id = int(fields.get("pattern_id", self._pattern_id))
117 elif kind == int(ck.START_REPLAY):
118 self._transmitting = True
119 self._enabled = True
120 ch = int(fields.get("target_channel", self._active_channel))
121 if ch != 0xFFFFFFFF: # REPLAY_KEEP_ORIGINAL -> Kanal unveraendert
122 self._active_channel = ch
123 elif kind == int(ck.STOP_TRANSMIT):
124 self._transmitting = False
125 self._enabled = False
126 # PING/SET_OVERFLOW_POLICY/SHUTDOWN: keine sichtbare Telemetrie-Wirkung.
127 self._applied_command_seq = command_seq
128
129 def jammer_state(self) -> tuple[bool, int, float]:
130 """Fuer die Kopplung mit dem Fake-Spektrum."""
131 return self._enabled, self._active_channel, self._tx_gain_db

153 def poll(self) -> list[dict[str, Any]]:
154 """Liefert die seit dem letzten Aufruf faelligen Records (aelteste zuerst).
155
156 Beruecksichtigt Halbduplex (waehrend TX gar nichts) und die Retune-Luecke
157 (schweigt bis ``_retune_until``, meldet danach die aufgeschobene Sequenz).
158 Sonst je Zyklus: Flankenereignisse, Occupancy, Regler-/Aktuatorzustand,
159 Latenz, gelegentlich SuccessMetric und BufferHealth.
160 """
161 now = time.monotonic()
162 if self._transmitting:
163 # Halbduplex: waehrend gesendet wird, steht der RX-Betrieb komplett.
164 # Wie der echte Core liefern wir hier NICHTS -- die Ack-Ebene 3 des
165 # START_JAMMING wird erst nach STOP_TRANSMIT sichtbar, wenn RX
166 # zurueckkommt. Das ist eine Sendephase, kein Datenausfall.
167 return []
168 if self._retune_until is not None:
169 if now < self._retune_until:
170 return [] # RX-Unterbrechung: der Core schweigt hier ebenfalls
171 self._retune_until = None
172 self._applied_command_seq = self._retune_seq
173 if now - self._last_cycle < _CYCLE_INTERVAL_S:
174 return []
175 self._last_cycle = now
176 t = now - self._t0
177
178 out: list[dict[str, Any]] = []
179 occupied, energy = self._simulate_occupancy(t)
180
181 for ch in sorted(occupied - self._prev_occupied):
182 out.append(self._detection(ch, tel.CHANNEL_EVENT_BECAME_OCCUPIED, energy[ch]))
183 for ch in sorted(self._prev_occupied - occupied):
184 out.append(self._detection(ch, tel.CHANNEL_EVENT_BECAME_FREE, energy[ch]))
185 self._prev_occupied = occupied
186
187 decision = self._advance_controller(occupied, energy)
188
189 out.append(self._occupancy(occupied, energy))
190 out.append(self._controller_state(decision))
191 out.append(self._actuator_state())
192 out.append(self._latency(t))
193
194 if self._enabled and now - self._last_success > 0.5:
195 self._last_success = now
196 out.append(self._success_metric())
197
198 if now - self._last_health > 1.0:
199 self._last_health = now
200 out.append(self._buffer_health())
201
202 return out

210 def _simulate_occupancy(self, t: float) -> tuple[set[int], dict[int, int]]:
211 """Wuerfelt Belegung + Energie je Kanal: ADV-Kanaele pulsieren, dazu ein
212 wandernder Hop-Kanal. Liefert (belegte Kanaele, energy[ch] in dBm)."""
213 occupied: set[int] = set()
214 energy = dict.fromkeys(range(40), -95)
215 for i, ch in enumerate(_ADV_CHANNELS):
216 if math.sin(t * 1.5 + i * 2.0) > 0.2:
217 occupied.add(ch)
218 energy[ch] = int(-45 + 8 * math.sin(t * 3 + i))
219 hop = int((t * 3) % 37)
220 occupied.add(hop)
221 energy[hop] = int(-38 + 5 * math.sin(t * 5))
222 return occupied, energy

224 def _advance_controller(self, occupied: set[int], energy: dict[int, int]) -> int:
225 """Simuliert einen Zyklus der Reglerlogik: waehlt den energiereichsten
226 belegten Kanal als Ziel und leitet daraus die Entscheidung ab
227 (ENABLE/RETUNE/HOLD/DISABLE). Gibt den Entscheidungs-Code zurueck."""
228 if not self._running or not occupied:
229 self._enabled = False
230 self._prev_target = None
231 return int(tel.CONTROLLER_DECISION_DISABLE)
232
233 target = max(occupied, key=lambda ch: energy[ch])
234 self._target_channel = target
235 self._active_channel = target
236 was_enabled = self._enabled
237 self._enabled = True
238
239 if not was_enabled:
240 decision = tel.CONTROLLER_DECISION_ENABLE
241 elif self._prev_target is not None and target != self._prev_target:
242 decision = tel.CONTROLLER_DECISION_RETUNE
243 else:
244 decision = tel.CONTROLLER_DECISION_HOLD
245 self._prev_target = target
246 return int(decision)

304 def _latency(self, t: float) -> dict[str, Any]:
305 """Baut den #4 latency-Record (Zeitstempel t1..t4 + µs-Segmente)."""
306 base = time.monotonic_ns()
307 fft = int(120_000 + 40_000 * abs(math.sin(t * 2)))
308 decision = int(30_000 + 15_000 * abs(math.cos(t * 3)))
309 dac = int(60_000 + 20_000 * abs(math.sin(t * 1.5)))
310 t1 = base
311 t2 = t1 + fft
312 t3 = t2 + decision
313 t4 = t3 + dac
314 return {
315 "kind": "latency",
316 "sequence": self._next_seq(),
317 "timestamp_ns": base,
318 "t1_adc_ns": t1,
319 "t2_fft_done_ns": t2,
320 "t3_decision_ns": t3,
321 "t4_dac_out_ns": t4,
322 "fft_us": fft / 1000.0,
323 "decision_us": decision / 1000.0,
324 "dac_us": dac / 1000.0,
325 "total_us": (t4 - t1) / 1000.0,
326 }
