1"""Synthetische Telemetrie fuer den Fake-Betrieb (ROD_FAKE=1).
2
3Erzeugt Records aller Typen und reagiert auf Kommandos: START/STOP schalten den
4Regelkreis, SET_TX_GAIN/SET_TARGET setzen Aktuator-Felder, und jedes angewandte
5Kommando spiegelt seine command_seq in applied_command_seq zurueck. Damit ist
6der Weg von pending nach applied (Ack-Ebene 3) auch ohne Core testbar.
7
8SuccessMetric (#7) liefert der Fake mit, obwohl der echte Core ihn derzeit gar
9nicht schreibt -- so laesst sich der UI-Pfad trotzdem ueben.
10
11Retunes sind absichtlich mit ihrer unangenehmen Seite nachgebaut: echte
12Hardware braucht Zeit, also schweigt der Fake fuer _RETUNE_GAP_S und meldet
13applied_command_seq erst danach. Waere das hier sofort fertig, wuerde die
14Wartelogik zum ersten Mal am echten Core getestet.
15"""

84 def apply_command(self, command_seq: int, kind: int, **fields: Any) -> None:
85 """Wendet ein validiertes Kommando auf den Sim-Zustand an."""
86 ck = tel_cmd()
87 if kind == int(ck.START):
88 self._running = True
89 elif kind == int(ck.STOP):
90 self._running = False
91 self._enabled = False
92 elif kind == int(ck.SET_TX_GAIN):
93 self._tx_gain_db = float(fields.get("tx_gain_db", self._tx_gain_db))
94 elif kind == int(ck.SET_TARGET):
95 self._pattern_id = int(fields.get("pattern_id", self._pattern_id))
96 elif kind in (int(ck.SET_SAMPLE_RATE), int(ck.SET_CENTER_FREQ)):
97 if kind == int(ck.SET_SAMPLE_RATE):
98 self._sample_rate_hz = float(fields.get("sample_rate_hz", self._sample_rate_hz))
99 else:
100 self._center_freq_hz = float(fields.get("center_freq_hz", self._center_freq_hz))
101 # Ack-Ebene 3 kommt erst nach der Unterbrechung. Ein zweiter Retune
102 # mitten in der Luecke faellt wie im Core mit dem ersten zusammen:
103 # die Luecke wird nicht laenger, nur die Sequenz rueckt vor.
104 now = time.monotonic()
105 if self._retune_until is None:
106 self._retune_until = now + _RETUNE_GAP_S
107 self._retune_seq = command_seq
108 return
109 elif kind == int(ck.START_RECORDING):
110 self._recording = True
111 elif kind == int(ck.STOP_RECORDING):
112 self._recording = False
113 elif kind == int(ck.START_JAMMING):
114 self._transmitting = True
115 self._enabled = True
116 self._active_channel = int(fields.get("target_channel", self._active_channel))
117 self._pattern_id = int(fields.get("pattern_id", self._pattern_id))
118 elif kind == int(ck.START_REPLAY):
119 self._transmitting = True
120 self._enabled = True
121 ch = int(fields.get("target_channel", self._active_channel))
122 if ch != 0xFFFFFFFF: # REPLAY_KEEP_ORIGINAL, Kanal bleibt wie er ist
123 self._active_channel = ch
124 elif kind == int(ck.STOP_TRANSMIT):
125 self._transmitting = False
126 self._enabled = False
127 elif kind == int(ck.START_SWEEP):
128 # Bereich und Verweildauer uebernehmen; 0 heisst BLE-Default
129 # 2402..2480 MHz bei 5 ms. Die Mitte marschiert dann in sdr_state().
130 start = float(fields.get("sweep_start_hz", 0)) or 2_402_000_000.0
131 end = float(fields.get("sweep_end_hz", 0)) or 2_480_000_000.0
132 dwell_us = int(fields.get("sweep_dwell_us", 0)) or 5000
133 self._sweep_start_hz = start
134 self._sweep_end_hz = end
135 self._sweep_dwell_s = max(0.001, dwell_us / 1e6)
136 self._sweep_t0 = time.monotonic()
137 self._sweep_active = True
138 elif kind == int(ck.STOP_SWEEP):
139 self._sweep_active = False
140 # PING, SET_OVERFLOW_POLICY und SHUTDOWN sieht man in der Telemetrie nicht.
141 self._applied_command_seq = command_seq

147 def sdr_state(self) -> tuple[float, float, bool]:
148 """Fenster und RX-Zustand fuer das Fake-Spektrum.
149
150 rx_paused deckt beide Faelle ab, in denen kein Frame fliesst: die kurze
151 Retune-Luecke und eine laufende Sendung. Waehrend eines Sweeps ist die
152 Mitte der gerade angefahrene 20-MHz-Schritt.
153 """
154 rx_paused = self._retune_until is not None or self._transmitting
155 center = self._sweep_center() if self._sweep_active else self._center_freq_hz
156 return center, self._sample_rate_hz, rx_paused
157
158 def _sweep_center(self) -> float:
159 # 20-MHz-Schritte zyklisch durch [start,end], je _sweep_dwell_s einer
160 # weiter. Grob so, wie es der echte Core macht.
161 window = 20_000_000.0
162 span = max(window, self._sweep_end_hz - self._sweep_start_hz)
163 n = max(1, round(span / window))
164 step = int((time.monotonic() - self._sweep_t0) / self._sweep_dwell_s) % n
165 return self._sweep_start_hz + window / 2 + step * window

180 def poll(self) -> list[dict[str, Any]]:
181 """Die seit dem letzten Aufruf faelligen Records, aelteste zuerst.
182
183 Pro Zyklus: Flankenereignisse, Occupancy, Regler- und Aktuatorzustand,
184 Latenz, dazu ab und an SuccessMetric und BufferHealth.
185 """
186 now = time.monotonic()
187 if self._transmitting:
188 # Halbduplex: waehrend gesendet wird, steht RX komplett. Wie der
189 # echte Core liefern wir hier nichts, die Ack-Ebene 3 des
190 # START_JAMMING wird erst nach STOP_TRANSMIT sichtbar. Das ist eine
191 # Sendephase und kein Datenausfall.
192 return []
193 if self._retune_until is not None:
194 if now < self._retune_until:
195 return [] # RX-Unterbrechung, der Core schweigt hier ebenfalls
196 self._retune_until = None
197 self._applied_command_seq = self._retune_seq
198 if now - self._last_cycle < _CYCLE_INTERVAL_S:
199 return []
200 self._last_cycle = now
201 t = now - self._t0
202
203 out: list[dict[str, Any]] = []
204 occupied, energy = self._simulate_occupancy(t)
205
206 for ch in sorted(occupied - self._prev_occupied):
207 out.append(self._detection(ch, tel.CHANNEL_EVENT_BECAME_OCCUPIED, energy[ch]))
208 for ch in sorted(self._prev_occupied - occupied):
209 out.append(self._detection(ch, tel.CHANNEL_EVENT_BECAME_FREE, energy[ch]))
210 self._prev_occupied = occupied
211
212 decision = self._advance_controller(occupied, energy)
213
214 out.append(self._occupancy(occupied, energy))
215 out.append(self._controller_state(decision))
216 out.append(self._actuator_state())
217 out.append(self._latency(t))
218
219 if self._enabled and now - self._last_success > 0.5:
220 self._last_success = now
221 out.append(self._success_metric())
222
223 if now - self._last_health > 1.0:
224 self._last_health = now
225 out.append(self._buffer_health())
226
227 return out

235 def _simulate_occupancy(self, t: float) -> tuple[set[int], dict[int, int]]:
236 # Die ADV-Kanaele pulsieren, dazu wandert ein Hop-Kanal durchs Band.
237 occupied: set[int] = set()
238 energy = dict.fromkeys(range(40), -95)
239 for i, ch in enumerate(_ADV_CHANNELS):
240 if math.sin(t * 1.5 + i * 2.0) > 0.2:
241 occupied.add(ch)
242 energy[ch] = int(-45 + 8 * math.sin(t * 3 + i))
243 hop = int((t * 3) % 37)
244 occupied.add(hop)
245 energy[hop] = int(-38 + 5 * math.sin(t * 5))
246 return occupied, energy

248 def _advance_controller(self, occupied: set[int], energy: dict[int, int]) -> int:
249 """Ein Zyklus Reglerlogik: energiereichster belegter Kanal wird Ziel.
250
251 Zurueck kommt der Entscheidungs-Code, ENABLE/RETUNE/HOLD/DISABLE.
252 """
253 if not self._running or not occupied:
254 self._enabled = False
255 self._prev_target = None
256 return int(tel.CONTROLLER_DECISION_DISABLE)
257
258 target = max(occupied, key=lambda ch: energy[ch])
259 self._target_channel = target
260 self._active_channel = target
261 was_enabled = self._enabled
262 self._enabled = True
263
264 if not was_enabled:
265 decision = tel.CONTROLLER_DECISION_ENABLE
266 elif self._prev_target is not None and target != self._prev_target:
267 decision = tel.CONTROLLER_DECISION_RETUNE
268 else:
269 decision = tel.CONTROLLER_DECISION_HOLD
270 self._prev_target = target
271 return int(decision)

326 def _latency(self, t: float) -> dict[str, Any]:
327 # Vier Zeitstempel t1..t4 und die Abschnitte dazwischen in Mikrosekunden.
328 base = time.monotonic_ns()
329 fft = int(120_000 + 40_000 * abs(math.sin(t * 2)))
330 decision = int(30_000 + 15_000 * abs(math.cos(t * 3)))
331 dac = int(60_000 + 20_000 * abs(math.sin(t * 1.5)))
332 t1 = base
333 t2 = t1 + fft
334 t3 = t2 + decision
335 t4 = t3 + dac
336 return {
337 "kind": "latency",
338 "sequence": self._next_seq(),
339 "timestamp_ns": base,
340 "t1_adc_ns": t1,
341 "t2_fft_done_ns": t2,
342 "t3_decision_ns": t3,
343 "t4_dac_out_ns": t4,
344 "fft_us": fft / 1000.0,
345 "decision_us": decision / 1000.0,
346 "dac_us": dac / 1000.0,
347 "total_us": (t4 - t1) / 1000.0,
348 }
