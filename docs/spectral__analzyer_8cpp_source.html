30SpectrumAnalyzer::SpectrumAnalyzer(std::uint32_t fft_size,
31 double sample_rate,
32 double center_freq,
33 WindowType window)
34 : fft_size_(fft_size),
35 sample_rate_(sample_rate),
36 center_freq_(center_freq) {
37
38 if (!isPowerOfTwo(fft_size_)) {
39 throw std::runtime_error("fft_size muss eine Zweierpotenz sein (Radix-2-FFT)");
40 }
41
42 real_.resize(fft_size_);
43 imag_.resize(fft_size_);
44 last_magnitudes_.assign(fft_size_, kFloorDb);
45
46 buildWindow(window);
47
48 // Bit-Reversal-Permutation vorberechnen: bit_rev_[i] ist der Index, mit
49 // dem i vor der iterativen FFT getauscht wird.
50 const std::uint32_t bits = log2u(fft_size_);
51 bit_rev_.resize(fft_size_);
52 for (std::uint32_t i = 0; i < fft_size_; ++i) {
53 std::uint32_t x = i;
54 std::uint32_t r = 0;
55 for (std::uint32_t b = 0; b < bits; ++b) {
56 r = (r << 1) | (x & 1u);
57 x >>= 1;
58 }
59 bit_rev_[i] = r;
60 }
61}

63void SpectrumAnalyzer::buildWindow(WindowType type) {
64 window_.resize(fft_size_);
65 const double N = static_cast<double>(fft_size_);
66 for (std::uint32_t n = 0; n < fft_size_; ++n) {
67 double w = 1.0;
68 switch (type) {
69 case WindowType::None:
70 w = 1.0;
71 break;
72 case WindowType::Hann:
73 w = 0.5 * (1.0 - std::cos(2.0 * kPi * n / (N - 1.0)));
74 break;
75 case WindowType::Hamming:
76 w = 0.54 - 0.46 * std::cos(2.0 * kPi * n / (N - 1.0));
77 break;
78 }
79 window_[n] = static_cast<float>(w);
80 }
81}

83void SpectrumAnalyzer::fftInPlace() {
84 const std::uint32_t n = fft_size_;
85
86 // 1) Bit-Reversal-Permutation anwenden.
87 for (std::uint32_t i = 0; i < n; ++i) {
88 const std::uint32_t j = bit_rev_[i];
89 if (j > i) {
90 std::swap(real_[i], real_[j]);
91 std::swap(imag_[i], imag_[j]);
92 }
93 }
94
95 // 2) Iterative Danielson-Lanczos-Stufen. Vorzeichen -1 im Exponenten =
96 // Vorwaerts-FFT.
97 for (std::uint32_t len = 2; len <= n; len <<= 1) {
98 const double ang = -2.0 * kPi / static_cast<double>(len);
99 const float wlen_r = static_cast<float>(std::cos(ang));
100 const float wlen_i = static_cast<float>(std::sin(ang));
101 for (std::uint32_t start = 0; start < n; start += len) {
102 float w_r = 1.0f;
103 float w_i = 0.0f;
104 for (std::uint32_t k = 0; k < len / 2; ++k) {
105 const std::uint32_t a = start + k;
106 const std::uint32_t b = start + k + len / 2;
107 const float u_r = real_[a];
108 const float u_i = imag_[a];
109 const float v_r = real_[b] * w_r - imag_[b] * w_i;
110 const float v_i = real_[b] * w_i + imag_[b] * w_r;
111 real_[a] = u_r + v_r;
112 imag_[a] = u_i + v_i;
113 real_[b] = u_r - v_r;
114 imag_[b] = u_i - v_i;
115 // w *= wlen
116 const float next_r = w_r * wlen_r - w_i * wlen_i;
117 const float next_i = w_r * wlen_i + w_i * wlen_r;
118 w_r = next_r;
119 w_i = next_i;
120 }
121 }
122 }

125bool SpectrumAnalyzer::process(const IqBlock& block, std::vector<float>& out_magnitudes_db) {
126 const std::size_t avail = block.num_samples;
127 const bool enough = avail >= fft_size_;
128
129 // Fenster anwenden und in die Arbeitspuffer laden. block.iq ist
130 // interleaved: iq[2i]=I, iq[2i+1]=Q.
131 for (std::uint32_t i = 0; i < fft_size_; ++i) {
132 if (i < avail) {
133 const float wi = window_[i];
134 real_[i] = block.iq[2 * i] * wi;
135 imag_[i] = block.iq[2 * i + 1] * wi;
136 } else {
137 // Zu wenige Samples -> mit Nullen auffuellen (Zero-Padding).
138 real_[i] = 0.0f;
139 imag_[i] = 0.0f;
140 }
141 }
142
143 fftInPlace();
144
145 // Magnituden in dB, fftshift-geordnet (negative Frequenzen zuerst, DC in
146 // der Mitte bei Index fft_size/2). Bin i der FFT gehoert zu:
147 // i < N/2 -> positive Frequenz
148 // i >= N/2 -> negative Frequenz
149 // Nach fftshift landet Bin (i + N/2) % N an Position i.
150 out_magnitudes_db.resize(fft_size_);
151 const float norm = 1.0f / static_cast<float>(fft_size_);
152 for (std::uint32_t i = 0; i < fft_size_; ++i) {
153 const std::uint32_t src = (i + fft_size_ / 2) % fft_size_;
154 const float re = real_[src] * norm;
155 const float im = imag_[src] * norm;
156 const float power = re * re + im * im;
157 // 10*log10(power) = 20*log10(magnitude). kFloorDb statt -inf bei 0.
158 out_magnitudes_db[i] = (power > 0.0f)
159 ? 10.0f * std::log10(power)
160 : kFloorDb;
161 }
162
163 last_magnitudes_ = out_magnitudes_db;
164 return enough;
165}

167void SpectrumAnalyzer::channelEnergies(std::vector<float>& out_channel_dbm) const {
168 out_channel_dbm.assign(kBleChannelCount, kFloorDb);
169
170 // Frequenz des Bins i (nach fftshift): center + (i - N/2) * (fs / N).
171 const double bin_hz = sample_rate_ / static_cast<double>(fft_size_);
172 const double first_freq = center_freq_ - (sample_rate_ / 2.0);
173
174 // Leistung wird zunächst pro FREQUENZ-INDEX gesammelt (physikalische
175 // Position, f = 2402 + 2k MHz) und erst danach auf die BLE-Kanalnummer
176 // abgebildet. Der frühere Code verwechselte beides und ordnete z.B. die
177 // Energie bei 2402 MHz fälschlich "Kanal 0" statt Kanal 37 zu -- siehe
178 // ble_channels.hpp für die (nicht-lineare) Zuordnung.
179 //
180 // Lineare Mittelung der Leistung, dann zurück in dB. In dB zu mitteln wäre
181 // falsch (arithmetisches Mittel von Logarithmen).
182 std::vector<double> power_sum(kBleChannelCount, 0.0);
183 std::vector<std::uint32_t> bin_count(kBleChannelCount, 0);
184
185 for (std::uint32_t i = 0; i < fft_size_; ++i) {
186 const double freq = first_freq + static_cast<double>(i) * bin_hz;
187 const double freq_mhz = freq / 1e6;
188
189 // Frequenz-Index k: (f - 2402) / 2, auf den nächsten Index gerundet.
190 const double idx_d = (freq_mhz - 2402.0) / 2.0;
191 const long idx = std::lround(idx_d);
192 if (idx < 0 || idx >= static_cast<long>(kBleChannelCount)) {
193 continue;
194 }
195 // Nur Bins nahe der Kanalmitte zählen (±0.5 Index = ±1 MHz).
196 if (std::abs(idx_d - static_cast<double>(idx)) > 0.5) {
197 continue;
198 }
199
200 // Frequenz-Index -> BLE-Kanalnummer (die eigentliche Korrektur).
201 const std::uint32_t ble_ch = freqIndexToBleChannel(static_cast<std::uint32_t>(idx));
202 if (ble_ch == kBleInvalidChannel) {
203 continue;
204 }
205
206 const float db = last_magnitudes_[i];
207 power_sum[ble_ch] += std::pow(10.0, db / 10.0);
208 ++bin_count[ble_ch];
209 }
210
211 for (std::uint32_t k = 0; k < kBleChannelCount; ++k) {
212 if (bin_count[k] > 0) {
213 const double mean_power = power_sum[k] / bin_count[k];
214 out_channel_dbm[k] = (mean_power > 0.0)
215 ? static_cast<float>(10.0 * std::log10(mean_power))
216 : kFloorDb;
217 }
218 }
219}
