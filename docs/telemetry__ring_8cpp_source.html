17TelemetryRing::TelemetryRing(const std::string& shm_name, std::uint64_t capacity_slots)
18 : shm_name_(shm_name) {
19
20 mapped_size_ = sizeof(RingHeader) + capacity_slots * sizeof(TelemetryRecord);
21
22 // Ein evtl. vorhandenes Altsegment (Leiche nach Core-Absturz) aus dem
23 // Namensraum nehmen. Fehler wird bewusst ignoriert: ENOENT ist der
24 // Normalfall (nichts aufzuräumen). Ein noch daran hängender
25 // Python-Prozess verliert sein Mapping dadurch NICHT -- das Segment
26 // wird erst freigegeben, wenn der letzte Nutzer es unmappt.
27 shm_unlink(shm_name_.c_str());
28
29 // O_EXCL: Wir bestehen auf einem frischen Segment. Schlägt das trotz
30 // des unlink oben fehl, hat ein zweiter Core-Prozess gerade parallel
31 // dasselbe Segment angelegt -- das ist ein Konfigurationsfehler, den
32 // wir laut melden statt uns gegenseitig die Daten zu überschreiben.
33 fd_ = shm_open(shm_name_.c_str(), O_CREAT | O_EXCL | O_RDWR, 0660);
34 if (fd_ < 0) {
35 throw std::runtime_error(
36 "shm_open fehlgeschlagen für " + shm_name_ +
37 " (laeuft bereits ein zweiter Core auf demselben Segment?)");
38 }
39
40 if (ftruncate(fd_, static_cast<off_t>(mapped_size_)) != 0) {
41 close(fd_);
42 throw std::runtime_error("ftruncate fehlgeschlagen für " + shm_name_);
43 }
44
45 mapped_base_ = mmap(nullptr, mapped_size_, PROT_READ | PROT_WRITE,
46 MAP_SHARED, fd_, 0);
47 if (mapped_base_ == MAP_FAILED) {
48 close(fd_);
49 throw std::runtime_error("mmap fehlgeschlagen für " + shm_name_);
50 }
51
52 header_ = static_cast<RingHeader*>(mapped_base_);
53 slots_ = reinterpret_cast<TelemetryRecord*>(
54 static_cast<std::uint8_t*>(mapped_base_) + sizeof(RingHeader));
55
56 // Header initialisieren. Dank O_EXCL oben ist das Segment garantiert
57 // frisch, kein Consumer kann es zu diesem Zeitpunkt bereits gemappt
58 // haben (der Name existierte bis eben nicht).
59 //
60 // Reihenfolge ist bindend: erst alle Nutzdaten-Felder, DANN magic.
61 header_->schema_version = kSchemaVersion;
62 header_->capacity_slots = capacity_slots;
63 header_->generation = static_cast<std::uint64_t>(
64 std::chrono::duration_cast<std::chrono::nanoseconds>(
65 std::chrono::system_clock::now().time_since_epoch()).count());
66 header_->write_index.store(0, std::memory_order_relaxed);
67 header_->read_index.store(0, std::memory_order_relaxed);
68 header_->dropped_count.store(0, std::memory_order_relaxed);
69
70 // Publikation: Release-Store stellt sicher, dass alle Schreibvorgänge
71 // oben für jeden Consumer sichtbar sind, der dieses magic liest.
72 // Ab hier ist das Segment für Python gültig -- keine Sekunde vorher.
73 header_->magic.store(kRingMagic, std::memory_order_release);
74}

87bool TelemetryRing::push(const TelemetryRecord& record) noexcept {
88 const std::uint64_t capacity = header_->capacity_slots;
89
90 // Acquire, weil wir gegen den read_index des Consumers prüfen -- wir
91 // müssen dessen bisherige Slot-Reads "sehen", bevor wir den Slot
92 // überschreiben.
93 const std::uint64_t read = header_->read_index.load(std::memory_order_acquire);
94 const std::uint64_t write = header_->write_index.load(std::memory_order_relaxed);
95
96 const std::uint64_t next_write = (write + 1) % capacity;
97 if (next_write == read) {
98 // Ring voll -- niemals blockieren, stattdessen zählen (Overflow-
99 // Policy: drop-newest). Der Drop-Counter fließt selbst als
100 // Buffer-Health-Telemetrie (#8) zurück.
101 header_->dropped_count.fetch_add(1, std::memory_order_relaxed);
102 return false;
103 }
104
105 // 1) Slot vollständig beschreiben, BEVOR der Index vorrückt.
106 slots_[write] = record;
107
108 // 2) Release-Store: garantiert, dass der obige Slot-Write für den
109 // Consumer sichtbar ist, sobald er den neuen write_index sieht.
110 header_->write_index.store(next_write, std::memory_order_release);
111
112 return true;
113}
