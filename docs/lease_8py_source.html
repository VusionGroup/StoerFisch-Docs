1"""Schreibrecht-Lease: genau EIN Writer, alle anderen sind Zuschauer.
2
3Wer als Erster Schreibrechte anfordert und noch keiner sie hat, wird Writer;
4weitere Anforderer landen in einer FIFO-Warteschlange. Der Writer verliert das
5Recht, wenn er es (1) abgibt (release), (2) seine Verbindung weg ist (owner_gone
6oder Heartbeat-Timeout) oder (3) auf eine Anwesenheits-Rueckfrage nicht reagiert
7-- dann rueckt der naechste noch verbundene Anforderer nach. Alle Steuer-
8Endpunkte pruefen gegen den aktuellen Writer; Zuschauer koennen nur lesen.
9
10Zwei Timeouts, die man auseinanderhalten muss:
11
12* **Verbindungs-Timeout** (``timeout_s``): naehrt sich aus dem AUTOMATISCHEN
13 WebSocket-Heartbeat (2 Hz). Faellt der aus, ist die Verbindung tot -> weg.
14* **Anwesenheits-Rueckfrage** (``presence_interval_s`` + ``challenge_grace_s``):
15 Sind ausser dem Writer noch andere verbunden und war er lange UNTAETIG (kein
16 Kommando, kein „noch da"-Klick), wird nachgefragt. Bleibt die Antwort aus,
17 wird das Recht weitergereicht. Der automatische Heartbeat zaehlt hier bewusst
18 NICHT als Anwesenheit -- sonst behielte ein AFK-Tab das Recht ewig.
19"""

113 def owner_gone(self, owner: str) -> None:
114 """Eine Verbindung ist weg: war es der Writer -> abgeben + befoerdern;

122 def poll_timeout(self) -> None:
123 """Prueft, ob der Writer verbindungstot ist (aus owners raus ODER
124 Heartbeat abgelaufen), und befoerdert dann den naechsten. (Backup zu
125 owner_gone fuer unsaubere Verbindungsabbrueche.)"""
126 if self._holder is None:
127 return
128 gone = self._holder not in self._owners
129 stale = (
130 self._last_seen is not None
131 and time.monotonic() - self._last_seen > self._timeout_s
132 )
133 if gone or stale:

137 def poll(self, subscriber_count: int) -> None:
138 """Periodischer Tick (aus dem TelemetryHub). Erst der Verbindungs-Timeout
139 (``poll_timeout``), dann die Anwesenheits-Rueckfrage:
140
141 Sind AUSSER dem Writer noch andere verbunden (``subscriber_count > 1``)
142 und war er laenger als ``presence_interval_s`` untaetig, wird eine
143 Rueckfrage gestartet. Reagiert er nicht binnen ``challenge_grace_s`` mit
144 ``confirm_presence``, verliert er das Recht an den naechsten Wartenden.
145 Ist er allein, wird NICHT nachgefragt -- allein stoert niemand.
146 """
147 self.poll_timeout()
148 if self._holder is None:
149 self._challenge_since = None
150 return
151 now = time.monotonic()
152 if self._challenge_since is not None:
153 # Rueckfrage laeuft: Frist abgelaufen ohne Bestaetigung -> weitergeben.
154 if now - self._challenge_since > self._challenge_grace_s:
155 self._clear_writer_state()
156 self._promote()
157 return
158 if (
159 subscriber_count > 1
160 and self._last_present is not None
161 and now - self._last_present > self._presence_interval_s
162 ):
163 self._challenge_since = now

165 def snapshot(self) -> LeaseSnapshot:
166 """Liefert eine unveraenderliche Momentaufnahme fuer REST/GUI."""
167 now = time.monotonic()
168 since = None if self._last_seen is None else now - self._last_seen
169 if self._challenge_since is None:
170 active, left = False, None
171 else:
172 active = True
173 left = max(0.0, self._challenge_grace_s - (now - self._challenge_since))
174 return LeaseSnapshot(
175 self._holder, list(self._queue), since, self._timeout_s, active, left
176 )

178 def _grant(self, owner: str) -> None:
179 """Vergibt das Schreibrecht an ``owner`` (interner Helfer). Frische
180 Anwesenheit, keine offene Rueckfrage."""
181 self._holder = owner
182 now = time.monotonic()
183 self._last_seen = now
184 self._last_present = now
185 self._challenge_since = None
186 if owner in self._queue:
187 self._queue.remove(owner)

196 def _promote(self) -> None:
197 """Befoerdert den naechsten noch verbundenen Anforderer zum Writer."""
198 while self._queue:
199 nxt = self._queue.pop(0)
