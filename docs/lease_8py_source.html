1"""Schreibrecht-Lease: genau einer darf steuern, alle anderen schauen zu.
2
3Wer zuerst anfragt und noch keinen Writer vorfindet, wird Writer; alle weiteren
4landen in einer FIFO-Warteschlange. Das Recht ist wieder weg, wenn der Writer es
5abgibt, seine Verbindung stirbt oder er auf die Anwesenheits-Rueckfrage nicht
6reagiert -- dann rueckt der naechste noch verbundene Anforderer nach. Die
7Steuer-Endpunkte pruefen alle gegen den aktuellen Writer, Zuschauer duerfen nur
8lesen.
9
10Wichtig ist, die beiden Timeouts auseinanderzuhalten. Der Verbindungs-Timeout
11(timeout_s) haengt am automatischen WebSocket-Heartbeat mit 2 Hz: bleibt der
12aus, ist die Verbindung tot. Die Anwesenheits-Rueckfrage
13(presence_interval_s plus challenge_grace_s) greift, wenn ausser dem Writer noch
14jemand verbunden ist und der Writer lange nichts getan hat -- kein Kommando,
15kein "noch da"-Klick. Bleibt die Antwort aus, wandert das Recht weiter. Der
16automatische Heartbeat zaehlt dabei bewusst nicht als Anwesenheit, sonst
17behielte ein vergessener Tab das Schreibrecht fuer immer.
18"""

110 def poll_timeout(self) -> None:
111 """Raeumt einen verbindungstoten Writer weg und befoerdert den naechsten.
112
113 Tot heisst: nicht mehr in owners oder Heartbeat abgelaufen. Das ist die
114 Absicherung fuer Verbindungsabbrueche, bei denen owner_gone ausbleibt.
115 """
116 if self._holder is None:
117 return
118 gone = self._holder not in self._owners
119 stale = (
120 self._last_seen is not None
121 and time.monotonic() - self._last_seen > self._timeout_s
122 )
123 if gone or stale:

127 def poll(self, subscriber_count: int) -> None:
128 """Periodischer Tick aus dem TelemetryHub.
129
130 Erst der Verbindungs-Timeout, dann die Anwesenheits-Rueckfrage: sind
131 ausser dem Writer noch andere verbunden und war er laenger als
132 presence_interval_s untaetig, fragen wir nach. Kommt binnen
133 challenge_grace_s kein confirm_presence, geht das Recht an den naechsten
134 Wartenden. Ist der Writer allein, fragen wir gar nicht erst -- allein
135 stoert er niemanden.
136 """
137 self.poll_timeout()
138 if self._holder is None:
139 self._challenge_since = None
140 return
141 now = time.monotonic()
142 if self._challenge_since is not None:
143 # Rueckfrage laeuft. Frist um und nichts gekommen, also weitergeben.
144 if now - self._challenge_since > self._challenge_grace_s:
145 self._clear_writer_state()
146 self._promote()
147 return
148 if (
149 subscriber_count > 1
150 and self._last_present is not None
151 and now - self._last_present > self._presence_interval_s
152 ):
153 self._challenge_since = now

155 def snapshot(self) -> LeaseSnapshot:
156 now = time.monotonic()
157 since = None if self._last_seen is None else now - self._last_seen
158 if self._challenge_since is None:
159 active, left = False, None
160 else:
161 active = True
162 left = max(0.0, self._challenge_grace_s - (now - self._challenge_since))
163 return LeaseSnapshot(
164 self._holder, list(self._queue), since, self._timeout_s, active, left
165 )

168 # Frisch befoerdert heisst: Anwesenheit gilt als bestaetigt, keine
