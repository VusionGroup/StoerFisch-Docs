35async def _open_real_sources(config: Config) -> tuple[Any, Any]:
36 """Oeffnet Telemetrie- und Spektral-Ring mit Retry (Core startet evtl. noch)."""
37 import spectral_layout as spec
38 import telemetry_layout as tel
39
40 from rod.sources.shm_spectral import ShmSpectralSource
41 from rod.sources.shm_telemetry import ShmTelemetrySource
42
43 tel_source = None
44 spec_source = None
45 for _ in range(200):
46 try:
47 if tel_source is None:
48 tel_source = ShmTelemetrySource(config.shm_telemetry)
49 if spec_source is None:
50 spec_source = ShmSpectralSource(config.shm_spectral)
51 break
52 except (tel.RingNotReadyError, spec.SpectralRingNotReadyError):
53 await asyncio.sleep(0.05)
54 else:
55 raise RuntimeError(
56 "Core nicht erreichbar: Ringe wurden nicht bereit. "
57 "Laeuft vusion_core? Ggf. ROD_SHM_* / ROD_FAKE pruefen."
58 )
59 return tel_source, spec_source
