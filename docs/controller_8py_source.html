1"""Die Steuerlogik zwischen REST und Kommando-Transport.
2
3Haelt den streng monoton steigenden command_seq-Zaehler -- das ist unsere
4Aufgabe, der Core prueft die Monotonie nicht --, baut die Frames ueber
5command.py, besteht auf status == OK und meldet die Sequenz beim PendingTracker
6an. Das Scharfschalten haengt bewusst an START und STOP: arm schickt START,
7disarm und der Dead-Man schicken STOP.
8
9Die SDR-Konfiguration ist inzwischen ein echtes Kommando. Frueher kam sie nur
10per CLI und SdrConfig war reine Anzeige, seit SET_SAMPLE_RATE und
11SET_CENTER_FREQ stellt der Core die Hardware zur Laufzeit um. Das hat zwei
12Folgen, die hier abgebildet sind: ein Retune unterbricht RX fuer einige
13Millisekunden, in denen weder Frames noch Telemetrie kommen -- normal, kein
14toter Core. Und die Ack-Ebene 3 kommt spaeter als sonst, weil der Core
15applied_command_seq erst meldet, wenn der Stream wieder laeuft. Wir uebernehmen
16die neue Sicht deshalb schon bei status == OK und lassen den PendingTracker
17nachziehen; solange steht das UI auf "pending".
18"""

118 async def _send(
119 self, kind: cmd.CommandKind, label: str, **fields: Any,
120 ) -> cmd.CommandResponse:
121 """Der zentrale Sendepfad, durch den jedes Kommando laeuft."""
122 seq = self._next_seq()
123 frame = cmd.build_request(kind, command_seq=seq, **fields)
124 response = await self._transport.request(frame)
125 response.raise_for_status() # CommandError, sobald status != OK
126 self._pending.register_requested(seq, label=label)
127 return response

183 async def set_sdr(self, sdr: SdrConfig) -> list[cmd.CommandResponse]:
184 """Stellt Samplerate und Center-Frequenz am Core ein.
185
186 Gesendet wird nur, was sich gegenueber der aktuellen Sicht wirklich
187 geaendert hat. Jedes Kommando kostet einen echten Retune samt RX-Luecke,
188 ein unveraendertes Feld noch einmal zu setzen bringt also nur die Luecke.
189 """
190 responses: list[cmd.CommandResponse] = []
191 if sdr.sample_rate_hz != self._sdr.sample_rate_hz:
192 responses.append(await self.set_sample_rate(sdr.sample_rate_hz))
193 if sdr.center_freq_hz != self._sdr.center_freq_hz:
194 responses.append(await self.set_center_freq(sdr.center_freq_hz))
195 return responses

197 # ---- Frequenz-Sweep (reiner RX, kein Scharfschalten noetig) --------------
198 #
199 # Beim Sweep faehrt die Mittenfrequenz zyklisch von start_hz bis end_hz und
200 # verweilt je Schritt dwell_us. So deckt der HackRF ueber die Zeit ein
201 # breiteres Band ab, als sein 20-MHz-Fenster auf einmal zeigt; gedacht ist
202 # das fuer die volle BLE-Bandbreite von 2402 bis 2480 MHz. Die Hardware
203 # stimmt dabei laufend um, die Center-Frequenz eines Frames steht also
204 # weiterhin im Frame selbst -- die Rod-Sicht kennt hier keinen festen Wert.
205
206 async def start_sweep(
207 self, start_hz: float, end_hz: float, dwell_us: int,
208 ) -> cmd.CommandResponse:
209 """Faehrt die Mittenfrequenz zyklisch von start_hz bis end_hz.
210
211 Bei Schema 1 senden wir gar nicht erst: die Sweep-Felder passen dort
212 nicht in den 40-Byte-Frame.
213 """
214 if not cmd.SWEEP_SUPPORTED:
215 raise SweepUnsupportedError(
216 "Frequenz-Sweep erst ab Command-Schema 2 verfuegbar "
217 f"(Core spricht Schema {cmd.COMMAND_SCHEMA_VERSION})."
218 )
219 return await self._send(
220 cmd.CommandKind.START_SWEEP,
221 f"sweep {start_hz / 1e6:g}..{end_hz / 1e6:g}MHz @{dwell_us}us",
222 sweep_start_hz=int(round(start_hz)),
223 sweep_end_hz=int(round(end_hz)),
224 sweep_dwell_us=int(dwell_us),
225 )
226

249 # ---- Sendebetrieb: Jamming und Replay, halbduplex ------------------------
250 #
251 # Der HackRF kann nicht gleichzeitig senden und empfangen. Sobald Jamming
252 # oder Replay laeuft, steht RX still: keine Frames, keine Telemetrie, bis
253 # STOP_TRANSMIT kommt. Das ist kein Datenausfall, sondern eine Sendephase.
254 # Entsprechend wird die Ack-Ebene 3 des START_JAMMING erst danach sichtbar,
255 # weil der TelemetryRing waehrend TX schweigt.
256 #
257 # Scharfschalten ist Voraussetzung, siehe TxSafetyError.
258
259 async def start_jamming(
260 self, target_channel: int, pattern_id: int = JAMMING_PATTERN_CARRIER,
261 ) -> cmd.CommandResponse:
262 """Blockiert einen Kanal aktiv. RX steht ab jetzt still."""
263 if not self._safety.armed:
264 raise TxSafetyError("TX nicht scharfgeschaltet -- erst arm_tx()")
265 return await self._send(
266 cmd.CommandKind.START_JAMMING, f"jam ch{target_channel} pat{pattern_id}",
267 target_channel=target_channel, pattern_id=pattern_id,
268 )

270 async def start_replay(
271 self, target_channel: int | None = None, loop: bool = False,
272 ) -> cmd.CommandResponse:
273 """Spielt die letzte Aufnahme ab. RX steht ab jetzt still.
274
275 Ohne target_channel bleibt die Aufnahmefrequenz stehen, sonst wandert
276 die Waveform auf den angegebenen Kanal. Mit loop laeuft sie bis
277 STOP_TRANSMIT, sonst geht der Core am Dateiende von selbst zurueck auf RX.
278 """
279 if not self._safety.armed:
280 raise TxSafetyError("TX nicht scharfgeschaltet -- erst arm_tx()")
281 channel = REPLAY_KEEP_ORIGINAL if target_channel is None else target_channel
282 label = "replay/original" if target_channel is None else f"replay ch{target_channel}"
283 return await self._send(
284 cmd.CommandKind.START_REPLAY, f"{label} loop={loop}",
285 target_channel=channel, pattern_id=1 if loop else 0,
286 )

307 async def disarm_tx(self) -> cmd.CommandResponse:
308 """Stoppt Sendung und Regelkreis, entschaerft.
309
310 STOP_TRANSMIT kommt zuerst und nur best effort: Entschaerfen muss eine
311 laufende Sendung beenden und nicht bloss den Regelkreis. Da das Kommando
312 idempotent ist, laeuft es ins Leere, wenn gerade nicht gesendet wird.
313 """
314 try:
315 with contextlib.suppress(Exception):
316 await self._send(cmd.CommandKind.STOP_TRANSMIT, "disarm/tx-stop")
317 return await self._send(cmd.CommandKind.STOP, "disarm/stop")
318 finally:
319 self._safety.disarm()

321 async def emergency_stop(self) -> None:
322 """Vom Dead-Man ausgeloest: Sendung und Regelkreis stoppen, entschaerfen.
323
324 Wieder STOP_TRANSMIT zuerst. Ein unbeaufsichtigter Sender ist genau der
325 Fall, fuer den es den Dead-Man gibt -- nur den Regelkreis zu stoppen
326 wuerde laufendes Jamming nicht beenden. Beides best effort.
327 """
328 for kind, label in (
329 (cmd.CommandKind.STOP_TRANSMIT, "deadman/tx-stop"),
330 (cmd.CommandKind.STOP, "deadman/stop"),
331 ):
332 with contextlib.suppress(Exception):
333 await self._send(kind, label)
334 self._safety.disarm()
