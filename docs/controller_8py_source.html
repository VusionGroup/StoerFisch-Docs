1"""Controller: die Steuerlogik zwischen REST und dem Kommando-Transport.
2
3Haelt den streng monotonen ``command_seq``-Zaehler (unsere Verantwortung, der
4Core prueft die Monotonie nicht), baut die Frames ueber command.py, prueft
5``status == OK`` und meldet die Sequenz beim PendingTracker an. Das Scharf-
6schalten ist bewusst an START/STOP gekoppelt: ``arm`` -> START, ``disarm`` und
7der Dead-Man -> STOP.
8
9SDR-KONFIGURATION IST JETZT EIN KOMMANDO: Seit COMMAND_SCHEMA_VERSION 1 mit
10SET_SAMPLE_RATE (7) / SET_CENTER_FREQ (8) stellt der Core die HackRF-Hardware
11zur Laufzeit um; frueher kam beides nur per CLI und ``SdrConfig`` war reine
12Anzeige. Zwei Folgen daraus, die hier bewusst abgebildet sind:
13
14* Ein Retune unterbricht RX fuer einige Millisekunden -- waehrenddessen kommen
15 weder Spektral-Frames noch Telemetrie. Das ist normal, kein toter Core.
16* Ack-Ebene 3 kommt spaeter als bei allen anderen Kommandos: der Core meldet
17 ``applied_command_seq`` erst, wenn der Stream wieder laeuft. Wir uebernehmen
18 die neue Sicht deshalb erst nach ``status == OK`` und lassen den
19 PendingTracker die Ebene 3 nachziehen -- solange steht das UI auf "pending".
20"""

71class TxSafetyError(RuntimeError):
72 """Aktives Senden wurde ohne scharfgeschalteten Owner angefordert.
73
74 Anders als frueher ist Senden echt (Jamming/Replay stellen den HackRF auf
75 TX). Der Dead-Man-Switch existiert genau dafuer -- deshalb ist scharfgeschaltet

81class Controller:
82 """Uebersetzt hochsprachliche Aktionen (arm, jam, retune, ...) in Kommando-
83 Frames, verschickt sie ueber den Transport, prueft den Status und meldet die
84 Sequenz beim PendingTracker an. Haelt die aktuelle SDR-/Jammer-Sicht und den
85 monotonen ``command_seq``-Zaehler."""

113 async def _send(
114 self, kind: cmd.CommandKind, label: str, **fields: Any,
115 ) -> cmd.CommandResponse:
116 """Zentraler Sendepfad: neue Sequenz ziehen, Frame bauen, senden,
117 ``status == OK`` erzwingen (wirft sonst CommandError) und die Sequenz
118 beim PendingTracker als angefordert vermerken."""
119 seq = self._next_seq()
120 frame = cmd.build_request(kind, command_seq=seq, **fields)
121 response = await self._transport.request(frame)
122 response.raise_for_status() # wirft CommandError bei status != OK
123 self._pending.register_requested(seq, label=label)
124 return response

155 # ---- SDR / Retune --------------------------------------------------------
156 #
157 # Beide Kommandos sind absolut und idempotent; ein Retry nach Timeout ist
158 # gefahrlos. Der Core fasst sie zu EINEM Retune zusammen, wenn das zweite
159 # eintrifft, bevor das erste ausgefuehrt wurde -- deshalb senden wir sie

180 async def set_sdr(self, sdr: SdrConfig) -> list[cmd.CommandResponse]:
181 """Stellt Samplerate und Center-Frequenz am Core ein.
182
183 Es werden nur die Felder gesendet, die sich gegenueber der aktuellen
184 Sicht tatsaechlich geaendert haben: Jedes Kommando kostet einen echten
185 HW-Retune samt RX-Luecke, und ein unveraendertes Feld noch einmal zu
186 setzen wuerde die Luecke ohne Nutzen erzeugen.
187 """
188 responses: list[cmd.CommandResponse] = []
189 if sdr.sample_rate_hz != self._sdr.sample_rate_hz:
190 responses.append(await self.set_sample_rate(sdr.sample_rate_hz))
191 if sdr.center_freq_hz != self._sdr.center_freq_hz:
192 responses.append(await self.set_center_freq(sdr.center_freq_hz))
193 return responses

210 # ---- Sendebetrieb: Jamming / Replay (HALBDUPLEX!) ------------------------
211 #
212 # Der HackRF kann nicht gleichzeitig senden und empfangen. Sobald Jamming
213 # oder Replay laeuft, STEHT der RX-Betrieb -- keine Spektral-Frames, keine
214 # Telemetrie, bis STOP_TRANSMIT. Das ist kein Datenausfall, sondern eine
215 # bewusste Sendephase. Ack-Ebene 3 des START_JAMMING wird deshalb erst nach
216 # STOP_TRANSMIT sichtbar (der TelemetryRing schweigt waehrend TX).
217 #
218 # Scharfschalten ist Voraussetzung: Senden ist echt, der Dead-Man muss es
219 # abdecken (siehe TxSafetyError).
220
221 async def start_jamming(
222 self, target_channel: int, pattern_id: int = JAMMING_PATTERN_CARRIER,
223 ) -> cmd.CommandResponse:
224 """START_JAMMING -- blockiert einen Kanal aktiv. RX steht ab jetzt still."""
225 if not self._safety.armed:
226 raise TxSafetyError("TX nicht scharfgeschaltet -- erst arm_tx()")
227 return await self._send(
228 cmd.CommandKind.START_JAMMING, f"jam ch{target_channel} pat{pattern_id}",
229 target_channel=target_channel, pattern_id=pattern_id,
230 )

232 async def start_replay(
233 self, target_channel: int | None = None, loop: bool = False,
234 ) -> cmd.CommandResponse:
235 """START_REPLAY -- spielt die letzte Aufnahme. RX steht ab jetzt still.
236
237 ``target_channel=None`` behaelt die Aufnahmefrequenz bei
238 (REPLAY_KEEP_ORIGINAL); sonst wird die Waveform auf den Kanal verschoben.
239 ``loop=True`` wiederholt bis STOP_TRANSMIT; sonst schaltet der Core nach
240 dem Dateiende von selbst in den RX-Betrieb zurueck.
241 """
242 if not self._safety.armed:
243 raise TxSafetyError("TX nicht scharfgeschaltet -- erst arm_tx()")
244 channel = REPLAY_KEEP_ORIGINAL if target_channel is None else target_channel
245 label = "replay/original" if target_channel is None else f"replay ch{target_channel}"
246 return await self._send(
247 cmd.CommandKind.START_REPLAY, f"{label} loop={loop}",
248 target_channel=channel, pattern_id=1 if loop else 0,
249 )

269 async def disarm_tx(self) -> cmd.CommandResponse:
270 """Stoppt Sendung UND Regelkreis, entschaerft.
271
272 STOP_TRANSMIT zuerst (best effort): Entschaerfen muss eine laufende
273 Sendung beenden, nicht nur den Regelkreis. STOP_TRANSMIT ist idempotent,
274 laeuft also auch ins Leere, wenn gerade nicht gesendet wird."""
275 try:
276 with contextlib.suppress(Exception):
277 await self._send(cmd.CommandKind.STOP_TRANSMIT, "disarm/tx-stop")
278 return await self._send(cmd.CommandKind.STOP, "disarm/stop")
279 finally:
280 self._safety.disarm()

282 async def emergency_stop(self) -> None:
283 """Vom Dead-Man ausgeloest: Sendung UND Regelkreis stoppen, entschaerfen.
284
285 STOP_TRANSMIT zuerst -- ein unbeaufsichtigter Sender ist genau der Fall,
286 fuer den der Dead-Man existiert; nur den Regelkreis zu stoppen wuerde ein
287 laufendes Jamming nicht beenden. Beide best effort."""
288 for kind, label in (
289 (cmd.CommandKind.STOP_TRANSMIT, "deadman/tx-stop"),
290 (cmd.CommandKind.STOP, "deadman/stop"),
291 ):
292 with contextlib.suppress(Exception):
293 await self._send(kind, label)
294 self._safety.disarm()
