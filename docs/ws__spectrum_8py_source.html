1"""WebSocket /ws/spectrum: die quantisierten Frames, binaer.
2
3Beim Verbinden geht einmal spectrum_meta als JSON raus, mit fft_size,
4sample_rate, Fenster und Qualitaetszaehlern. Danach kommt zu jedem neuen Frame
5ein Binaerpaket: 8 Byte float64 center, dann N Byte uint8-Bins. Die Bins
6zeichnet der Client direkt aufs Canvas und ausdruecklich nicht ueber den
7React-State, der ist fuer die langsamen Daten da.
8
9Etwa einmal pro Sekunde geht meta erneut raus, damit ein geaendertes Fenster,
10ein gewechselter center-Wert und die Verlustzaehler aktuell bleiben.
11"""

24async def spectrum_ws(websocket: WebSocket) -> None:
25 # Geschickt wird nur, wenn sich die Sequenz geaendert hat -- denselben Frame
26 # zweimal zu senden bringt niemandem etwas.
27 await websocket.accept()

30 interval = 1.0 / max(1.0, config.spectrum_fps)
31
32 await websocket.send_json(spectrum.meta())
33
34 last_seq = -1
35 meta_accum = 0.0

40 break
41
42 seq = spectrum.latest_seq
43 payload = spectrum.latest_payload
44 if payload is not None and seq != last_seq:
45 await websocket.send_bytes(payload)
46 last_seq = seq
47
48 meta_accum += interval
49 if meta_accum >= 1.0:
50 meta_accum = 0.0
51 await websocket.send_json(spectrum.meta())
52 except Exception as exc:
53 _logger.warning("spectrum_ws: Verbindung beendet durch %r", exc)
